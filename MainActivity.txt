class VaccineReminderReceiver : BroadcastReceiver() {
    override fun onReceive(context: Context, intent: Intent) {
        val channelId = "sheep_vaccination_reminders"
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val manager = context.getSystemService(Context.NOTIFICATION_SERVICE) as android.app.NotificationManager
            manager.createNotificationChannel(android.app.NotificationChannel(channelId, "تنبيهات التطعيمات", android.app.NotificationManager.IMPORTANCE_HIGH))
        }
        val notification = NotificationCompat.Builder(context, channelId)
            .setSmallIcon(android.R.drawable.ic_dialog_info)
            .setContentTitle(intent.getStringExtra("title") ?: "موعد تطعيم")
            .setContentText(intent.getStringExtra("body") ?: "راجع برنامج التطعيم")
            .setPriority(NotificationCompat.PRIORITY_HIGH).setAutoCancel(true).build()
        try { NotificationManagerCompat.from(context).notify(intent.getIntExtra("notificationId", 1001), notification) }
        catch (_: SecurityException) { }
    }
}

private fun vaccineAlarmCode(tag: String, key: String, label: String): Int =
    (tag + "|" + key + "|" + label).hashCode()

private fun cancelVaccineAlarm(context: Context, tag: String, key: String, label: String) {
    val code = vaccineAlarmCode(tag, key, label)
    val intent = Intent(context, VaccineReminderReceiver::class.java)
    val pending = PendingIntent.getBroadcast(
        context, code, intent,
        PendingIntent.FLAG_NO_CREATE or PendingIntent.FLAG_IMMUTABLE
    ) ?: return
    (context.getSystemService(Context.ALARM_SERVICE) as AlarmManager).cancel(pending)
    pending.cancel()
}

private fun scheduleVaccineAlarm(context: Context, tag: String, key: String, title: String, due: LocalDate, day: LocalDate, label: String) {
    val alarm = context.getSystemService(Context.ALARM_SERVICE) as AlarmManager
    val code = vaccineAlarmCode(tag, key, label)
    val intent = Intent(context, VaccineReminderReceiver::class.java).apply {
        putExtra("title", "تذكير تطعيم: " + tag)
        putExtra("body", title + " — " + label + ". موعد الاستحقاق: " + due)
        putExtra("notificationId", code)
    }
    val pending = PendingIntent.getBroadcast(context, code, intent, PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE)
    val millis = LocalDateTime.of(day, java.time.LocalTime.of(8, 0)).atZone(ZoneId.systemDefault()).toInstant().toEpochMilli()
    if (millis <= System.currentTimeMillis()) {
        alarm.cancel(pending)
        pending.cancel()
        return
    }
    try {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S && alarm.canScheduleExactAlarms())
            alarm.setExactAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, millis, pending)
        else alarm.setAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, millis, pending)
    } catch (_: SecurityException) { alarm.setAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, millis, pending) }
}

class VaccineBootReceiver : BroadcastReceiver() {
    override fun onReceive(context: Context, intent: Intent) {
        if (intent.action != Intent.ACTION_BOOT_COMPLETED && intent.action != Intent.ACTION_MY_PACKAGE_REPLACED) return
        val pending = goAsync()
        Thread {
            try {
                kotlinx.coroutines.runBlocking {
                    val latest = FarmDatabase.get(context).farmDao().vaccineRecords().first()
                        .filter { it.nextDueDate.isNotBlank() }
                        .groupBy { it.animalTag to it.vaccineKey }
                        .values.mapNotNull { records -> records.maxByOrNull { it.administeredDate + "|" + it.id } }
                    latest.forEach { record ->
                        val due = runCatching { LocalDate.parse(record.nextDueDate) }.getOrNull() ?: return@forEach
                        scheduleVaccineAlarm(context, record.animalTag, record.vaccineKey + "-next", record.vaccineName, due, due.minusDays(3), "تبقى 3 أيام على الجرعة التالية")
                        scheduleVaccineAlarm(context, record.animalTag, record.vaccineKey + "-next", record.vaccineName, due, due, "موعد الجرعة التالية المسجل")
                    }
                }
            } finally { pending.finish() }
        }.start()
    }
}

class FarmViewModel(private val dao: FarmDao) : ViewModel() {

    val animals = dao.activeAnimals().stateIn(
        viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList()
    )

    val purchases = dao.purchases().stateIn(
        viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList()
    )

    val consumptions = dao.consumptions().stateIn(
        viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList()
    )

    val tasks = dao.tasks().stateIn(
        viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList()
    )

    val events = dao.events().stateIn(
        viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList()
    )
    val vaccineRecords = dao.vaccineRecords().stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList())
    val vaccineCampaigns = dao.vaccineCampaigns().stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList())
    val careStatuses = dao.animalCareStatuses().stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList())

    fun addAnimal(
        tag: String,
        sex: String,
        category: String,
        weight: Double,
        onResult: (String?) -> Unit
    ) {
        viewModelScope.launch {
            try {
                dao.insertAnimal(
                    Animal(
                        tag = tag.trim(),
                        sex = sex,
                        category = category,
                        weightKg = weight
                    )
                )
                onResult(null)
            } catch (_: Exception) {
                onResult("تعذر الحفظ: تأكد أن رمز الحيوان غير مكرر.")
            }
        }
    }

    fun archive(animal: Animal, onResult: (Boolean) -> Unit) {
        viewModelScope.launch {
            try {
                dao.archiveAnimal(animal.tag, "أرشفة يدوية")
                onResult(true)
            } catch (_: Exception) {
                onResult(false)
            }
        }
    }

    fun updateAnimal(
        animal: Animal,
        onResult: (Boolean) -> Unit
    ) {
        viewModelScope.launch {
            try {
                dao.updateAnimal(animal)
                onResult(true)
            } catch (_: Exception) {
                onResult(false)
            }
        }
    }

    fun addAnimalEvent(
        tag: String,
        type: String,
        date: String,
        notes: String,
        onResult: (Boolean) -> Unit
    ) {
        viewModelScope.launch {
            try {
                dao.addEvent(
                    HerdEvent(
                        animalTag = tag,
                        eventType = type,
                        eventDate = date,
                        notes = notes
                    )
                )
                onResult(true)
            } catch (_: Exception) {
                onResult(false)
            }
        }
    }

    fun addPurchase(
        name: String,
        qty: Double,
        price: Double?,
        supplier: String,
        onResult: (Boolean) -> Unit
    ) {
        viewModelScope.launch {
            try {
                dao.addPurchase(
                    FeedPurchase(
                        feedName = name.trim(),
                        quantityKg = qty,
                        totalPriceDzd = price,
                        supplier = supplier,
                        purchaseDate = LocalDate.now().toString()
                    )
                )
                onResult(true)
            } catch (_: Exception) {
                onResult(false)
            }
        }
    }

    fun addConsumption(
        name: String,
        qty: Double,
        category: String,
        count: Int,
        notes: String,
        onResult: (Boolean) -> Unit
    ) {
        viewModelScope.launch {
            try {
                val saved = dao.addConsumptionSafely(
                    FeedConsumption(
                        feedName = name,
                        quantityKg = qty,
                        consumptionDate = LocalDate.now().toString(),
                        animalCategory = category,
                        animalCount = count,
                        notes = notes
                    )
                )
                onResult(saved)
            } catch (_: Exception) {
                onResult(false)
            }
        }
    }

    fun addTask(
        type: String,
        date: String,
        count: Int,
        notes: String,
        onResult: (Boolean) -> Unit
    ) {
        viewModelScope.launch {
            try {
                dao.addTask(
                    FarmTask(
                        taskType = type,
                        taskDate = date,
                        animalCount = count,
                        notes = notes
                    )
                )
                onResult(true)
            } catch (_: Exception) {
                onResult(false)
            }
        }
    }

    fun addVaccineRecord(record: VaccineDoseRecord, onResult: (Boolean) -> Unit) {
        viewModelScope.launch { onResult(runCatching { dao.addVaccineRecord(record) }.isSuccess) }
    }
    fun addVaccineCampaign(campaign: VaccineCampaign, onResult: (Boolean) -> Unit) {
        viewModelScope.launch { onResult(runCatching { dao.addVaccineCampaign(campaign) }.isSuccess) }
    }
    fun saveCareStatus(status: AnimalCareStatus, onResult: (Boolean) -> Unit) {
        viewModelScope.launch { onResult(runCatching { dao.saveAnimalCareStatus(status) }.isSuccess) }
    }
}

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun FarmScreen(vm: FarmViewModel) {
    val animals by vm.animals.collectAsState()
    val purchases by vm.purchases.collectAsState()
    val consumptions by vm.consumptions.collectAsState()
    val tasks by vm.tasks.collectAsState()
    val events by vm.events.collectAsState()
    val vaccineRecords by vm.vaccineRecords.collectAsState()
    val vaccineCampaigns by vm.vaccineCampaigns.collectAsState()
    val careStatuses by vm.careStatuses.collectAsState()

    var selectedTab by remember { mutableIntStateOf(0) }
    var selectedAnimal by remember { mutableStateOf<Animal?>(null) }
    var selectedSmartSection by remember { mutableStateOf<String?>(null) }
    var screenMessage by remember { mutableStateOf("") }

    val tabs = listOf(
        Triple("الرئيسية", "⌂", 0),
        Triple("القطيع", "🐑", 1),
        Triple("المشتريات", "🛒", 2),
        Triple("التغذية", "🌾", 3),
        Triple("الأقسام", "▦", 4)
    )

    Scaffold(
        containerColor = MaterialTheme.colorScheme.background,
        topBar = {
            TopAppBar(
                title = {
                    Column(verticalArrangement = Arrangement.spacedBy(0.dp)) {
                        Text(
                            if (selectedAnimal != null) "بطاقة الحيوان"
                            else selectedSmartSection ?: tabs[selectedTab].first,
                            style = MaterialTheme.typography.titleLarge
                        )
                        Text(
                            "مواشي • إدارة القطيع بذكاء",
                            style = MaterialTheme.typography.labelSmall,
                            color = MaterialTheme.colorScheme.onSurfaceVariant
                        )
                    }
                },
                colors = TopAppBarDefaults.topAppBarColors(
                    containerColor = MaterialTheme.colorScheme.surface
                )
            )
        },
        bottomBar = {
            NavigationBar(
                containerColor = MaterialTheme.colorScheme.surface,
                tonalElevation = 4.dp
            ) {
                tabs.forEachIndexed { index, item ->
                    NavigationBarItem(
                        selected = selectedTab == index && selectedAnimal == null,
                        onClick = {
                            selectedTab = index
                            selectedAnimal = null
                            selectedSmartSection = null
                            screenMessage = ""
                        },
                        icon = { FarmNavIcon(item.second, index, selectedTab == index) },
                        label = { Text(item.first, style = MaterialTheme.typography.labelSmall) }
                    )
                }
            }
        }
    ) { padding ->
        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(padding)
                .padding(horizontal = 12.dp, vertical = 8.dp)
        ) {
            if (selectedAnimal != null) {
                val animal = selectedAnimal!!
                AnimalDetailsScreen(
                    animal = animal,
                    events = events.filter { it.animalTag == animal.tag },
                    onBack = { selectedAnimal = null },
                    onUpdate = { updated ->
                        vm.updateAnimal(updated) { success ->
                            screenMessage = if (success) "تم تحديث البيانات بنجاح." else "تعذر تحديث البيانات."
                        }
                    },
                    onAddEvent = { type, date, notes ->
                        vm.addAnimalEvent(animal.tag, type, date, notes) { success ->
                            screenMessage = if (success) "تم تسجيل الحدث." else "تعذر تسجيل الحدث."
                        }
                    },
                    onArchive = {
                        vm.archive(animal) { success ->
                            if (success) selectedAnimal = null
                            else screenMessage = "تعذرت أرشفة الحيوان."
                        }
                    }
                )
                if (screenMessage.isNotBlank()) {
                    Text(
                        screenMessage,
                        modifier = Modifier.padding(8.dp),
                        color = MaterialTheme.colorScheme.primary
                    )
                }
            } else {
                when (selectedTab) {
                    0 -> HomeScreen(
                        animals, purchases, consumptions, tasks, events,
                        onNavigate = { selectedTab = it },
                        onAddEvent = { tag, type, date, notes, result ->
                            vm.addAnimalEvent(tag, type, date, notes, result)
                        },
                        onAddTask = { type, date, count, notes, result ->
                            vm.addTask(type, date, count, notes, result)
                        }
                    )
                    1 -> HerdScreen(
                        animals = animals,
                        onOpen = { selectedAnimal = it },
                        onAdd = { tag, sex, category, weight, result ->
                            vm.addAnimal(tag, sex, category, weight, result)
                        }
                    )
                    2 -> PurchasesScreen(
                        purchases = purchases,
                        onAdd = { name, qty, price, supplier, result ->
                            vm.addPurchase(name, qty, price, supplier, result)
                        }
                    )
                    3 -> FeedingScreen(
                        purchases = purchases,
                        consumptions = consumptions,
                        animals = animals,
                        onAdd = { name, qty, category, count, notes, result ->
                            vm.addConsumption(name, qty, category, count, notes, result)
                        }
                    )
                    else -> {
                        if (selectedSmartSection == null) {
                            SmartSectionsHome(onSelect = { selectedSmartSection = it })
                        } else {
                            SmartFarmPanel(
                                animals = animals,
                                purchases = purchases,
                                consumptions = consumptions,
                                tasks = tasks,
                                events = events,
                                vaccineRecords = vaccineRecords,
                                vaccineCampaigns = vaccineCampaigns,
                                careStatuses = careStatuses,
                                onAddVaccineRecord = { record, result -> vm.addVaccineRecord(record, result) },
                                onAddVaccineCampaign = { campaign, result -> vm.addVaccineCampaign(campaign, result) },
                                onSaveCareStatus = { status, result -> vm.saveCareStatus(status, result) },
                                initialSection = selectedSmartSection!!,
                                onBack = { selectedSmartSection = null },
                                onAddEvent = { tag, type, date, notes, result ->
                                    vm.addAnimalEvent(tag, type, date, notes, result)
                                },
                                onAddTask = { type, date, count, notes, result ->
                                    vm.addTask(type, date, count, notes, result)
                                }
                            )
                        }
                    }
                }
            }
        }
    }
}

@Composable

    animals: List<Animal>,
    tasks: List<FarmTask>,
    events: List<HerdEvent>

    val overdue = tasks.count { task ->
        !task.completed && runCatching { LocalDate.parse(task.taskDate) }.getOrNull()?.isBefore(today) == true
    }
    val dueSoon = tasks.count { task ->
        val date = runCatching { LocalDate.parse(task.taskDate) }.getOrNull()
        !task.completed && date != null && !date.isBefore(today) && !date.isAfter(today.plusDays(7))
    }

        val date = runCatching { LocalDate.parse(event.eventDate) }.getOrNull()

@Composable
fun FeedingScreen(
    purchases: List<FeedPurchase>,
    consumptions: List<FeedConsumption>,
    animals: List<Animal>,
    onAdd: (String, Double, String, Int, String, (Boolean) -> Unit) -> Unit
) {
    val feeds = listOf(
        "الشعير", "التبن", "النخالة",
        "الذرة", "البرسيم الأخضر", "البرسيم المجفف"
    )

    val categories = listOf(
        "القطيع كله", "نعاج", "حوامل",
        "مرضعات", "فحول", "حملان"
    )

    var feed by remember { mutableStateOf(feeds[0]) }
    var category by remember { mutableStateOf(categories[0]) }
    var quantity by remember { mutableStateOf("") }
    var count by remember(animals.size) {
        mutableStateOf(animals.size.toString())
    }
    var notes by remember { mutableStateOf("") }
    var message by remember { mutableStateOf("") }
    var saving by remember { mutableStateOf(false) }

    val balances = feeds.associateWith { name ->
        purchases.filter { it.feedName == name }.sumOf { it.quantityKg } -
            consumptions.filter { it.feedName == name }.sumOf { it.quantityKg }
    }

    Column(Modifier.fillMaxSize()) {

        Text("التغذية والمخزون", style = MaterialTheme.typography.headlineSmall)

        Text("رصيد الأعلاف", style = MaterialTheme.typography.titleMedium)

        LazyColumn(
            modifier = Modifier.heightIn(max = 190.dp),
            verticalArrangement = Arrangement.spacedBy(4.dp)
        ) {
            items(feeds) { name ->
                val balance = balances[name] ?: 0.0

                Card(Modifier.fillMaxWidth()) {
                    Row(
                        Modifier.fillMaxWidth().padding(10.dp),
                        horizontalArrangement = Arrangement.SpaceBetween
                    ) {
                        Text(name)
                        Text(
                            "${"%.2f".format(balance)} كغ",
                            color = if (balance <= 0)
                                MaterialTheme.colorScheme.error
                            else MaterialTheme.colorScheme.primary
                        )
                    }
                }
            }
        }

        HorizontalDivider()
        Text("تسجيل استهلاك", style = MaterialTheme.typography.titleMedium)

        DropdownField("نوع العلف", feed, feeds, { feed = it })
        OutlinedTextField(
            quantity, { quantity = it },
            label = { Text("الكمية المستهلكة بالكغ") },
            modifier = Modifier.fillMaxWidth(),
            enabled = !saving
        )

        DropdownField("الفئة", category, categories, { category = it })

        OutlinedTextField(
            count, { count = it },
            label = { Text("عدد الحيوانات") },
            modifier = Modifier.fillMaxWidth(),
            enabled = !saving
        )

        OutlinedTextField(
            notes, { notes = it },
            label = { Text("ملاحظات (اختياري)") },
            modifier = Modifier.fillMaxWidth(),
            enabled = !saving
        )

        Button(
            enabled = !saving,
            onClick = {
                val q = quantity.toDoubleOrNull()
                val c = count.toIntOrNull() ?: 0

                if (q == null || q <= 0 || q.isNaN() || q.isInfinite() || c <= 0) {
                    message = "أدخل كمية وعدداً صحيحين."
                } else {
                    saving = true
                    message = ""

                    onAdd(feed, q, category, c, notes) { success ->
                        saving = false
                        if (success) {
                            quantity = ""
                            notes = ""
                            message = "تم تسجيل الاستهلاك بنجاح."
                        } else {
                            message = "لم يتم التسجيل: الرصيد غير كافٍ أو حدث خطأ."
                        }
                    }
                }
            },
            modifier = Modifier.fillMaxWidth()
        ) {
            Text(if (saving) "جارٍ التحقق والحفظ..." else "تسجيل الاستهلاك")
        }

        if (message.isNotBlank()) {
            Text(
                message,
                color = if (message.startsWith("تم"))
                    MaterialTheme.colorScheme.primary
                else MaterialTheme.colorScheme.error
            )
        }

        val weekStart = LocalDate.now().minusDays(6)
        val recentEntries = consumptions.filter { entry ->
            entry.feedName == feed &&
                runCatching { LocalDate.parse(entry.consumptionDate) }.getOrNull()
                    ?.let { !it.isBefore(weekStart) && !it.isAfter(LocalDate.now()) } == true
        }
        val recentKg = recentEntries.sumOf { it.quantityKg }
        val recentHeadCount = recentEntries.sumOf { it.animalCount.coerceAtLeast(0) }
        Card(
            modifier = Modifier.fillMaxWidth(),
            colors = CardDefaults.cardColors(containerColor = MaterialTheme.colorScheme.secondaryContainer)
        ) {
            Column(Modifier.padding(12.dp), verticalArrangement = Arrangement.spacedBy(4.dp)) {
                Text("ملخص استهلاك $feed خلال 7 أيام", style = MaterialTheme.typography.titleSmall)
                Text("الكمية المسجلة: ${"%.2f".format(Locale.US, recentKg)} كغ")
                Text(if (recentHeadCount > 0)
                    "متوسط السجل لكل رأس: ${"%.3f".format(Locale.US, recentKg / recentHeadCount)} كغ"
                    else "لا توجد سجلات لهذا العلف خلال آخر 7 أيام.")
                Text("مؤشر حسابي للسجلات فقط، وليس حصة غذائية موصى بها.",
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant)
            }
        }

        HorizontalDivider()
        Text("سجل الاستهلاك", style = MaterialTheme.typography.titleMedium)

        LazyColumn(Modifier.weight(1f)) {
            items(consumptions, key = { it.id }) { item ->
                ListItem(
                    headlineContent = { Text(item.feedName) },
                    supportingContent = {
                        Text(
                            "${item.quantityKg} كغ - ${item.consumptionDate} - ${item.animalCategory}"
                        )
                    }
                )
                HorizontalDivider()
            }
        }
    }
}

@OptIn(ExperimentalMaterial3Api::class)
@Composable
private fun CalendarDateField(
    label: String,
    value: String,
    onDateSelected: (String) -> Unit
) {
    var showPicker by remember { mutableStateOf(false) }
    val parsedDate = remember(value) {
        runCatching { LocalDate.parse(value) }.getOrNull()
    }

    OutlinedTextField(
        value = parsedDate?.let {
            "%02d/%02d/%04d".format(Locale("ar"), it.dayOfMonth, it.monthValue, it.year)
        } ?: "",
        onValueChange = {},
        readOnly = true,
        label = { Text(label) },
        placeholder = { Text("اختر التاريخ من التقويم") },
        trailingIcon = {
            TextButton(onClick = { showPicker = true }) { Text("📅") }
        },
        modifier = Modifier.fillMaxWidth().clickable { showPicker = true }
    )

    if (showPicker) {
        val pickerState = rememberDatePickerState(
            initialSelectedDateMillis = parsedDate
                ?.atStartOfDay(ZoneOffset.UTC)?.toInstant()?.toEpochMilli()
        )
        DatePickerDialog(
            onDismissRequest = { showPicker = false },
            confirmButton = {
                TextButton(onClick = {
                    pickerState.selectedDateMillis?.let { millis ->
                        val selected = Instant.ofEpochMilli(millis)
                            .atZone(ZoneOffset.UTC).toLocalDate()
                        onDateSelected(selected.toString())
                    }
                    showPicker = false
                }) { Text("اختيار") }
            },
            dismissButton = {
                TextButton(onClick = { showPicker = false }) { Text("إلغاء") }
            }
        ) {
            DatePicker(state = pickerState, showModeToggle = true)
        }
    }
}

        OutlinedTextField(

        CalendarDateField(

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun DropdownField(
    label: String,
    value: String,
    options: List<String>,
    onSelect: (String) -> Unit,
    modifier: Modifier = Modifier
) {
    var expanded by remember { mutableStateOf(false) }

    ExposedDropdownMenuBox(
        expanded = expanded,
        onExpandedChange = { expanded = !expanded },
        modifier = modifier
    ) {
        OutlinedTextField(
            value = value,
            onValueChange = {},
            readOnly = true,
            label = { Text(label) },
            trailingIcon = {
                ExposedDropdownMenuDefaults.TrailingIcon(expanded = expanded)
            },
            modifier = Modifier.menuAnchor().fillMaxWidth()
        )

        ExposedDropdownMenu(
            expanded = expanded,
            onDismissRequest = { expanded = false }
        ) {
            options.forEach { option ->
                DropdownMenuItem(
                    text = { Text(option) },
                    onClick = {
                        onSelect(option)
                        expanded = false
                    }
                )
            }
        }
    }
}

@Composable
fun SmartFarmPanel(
    animals: List<Animal>,
    purchases: List<FeedPurchase>,
    consumptions: List<FeedConsumption>,
    tasks: List<FarmTask>,
    events: List<HerdEvent>,
    vaccineRecords: List<VaccineDoseRecord>,
    vaccineCampaigns: List<VaccineCampaign>,
    careStatuses: List<AnimalCareStatus>,
    onAddVaccineRecord: (VaccineDoseRecord, (Boolean) -> Unit) -> Unit,
    onAddVaccineCampaign: (VaccineCampaign, (Boolean) -> Unit) -> Unit,
    onSaveCareStatus: (AnimalCareStatus, (Boolean) -> Unit) -> Unit,
    initialSection: String,
    onBack: () -> Unit,
    onAddEvent: (String, String, String, String, (Boolean) -> Unit) -> Unit,
    onAddTask: (String, String, Int, String, (Boolean) -> Unit) -> Unit
) {
    val context = LocalContext.current
    val today = LocalDate.now()
    val section = initialSection
    var feedGroup by remember { mutableStateOf("القطيع كله") }
    var dailyPerHead by remember { mutableStateOf("") }
    var targetTag by remember(animals) { mutableStateOf(animals.firstOrNull()?.tag ?: "عام") }
    var eventType by remember { mutableStateOf("تطعيم") }
    var eventDate by remember { mutableStateOf(today.toString()) }
    var eventAmount by remember { mutableStateOf("") }
    var eventNotes by remember { mutableStateOf("") }
    var taskType by remember { mutableStateOf("فحص بيطري") }
    var taskDate by remember { mutableStateOf(today.plusDays(1).toString()) }
    var taskCount by remember(animals.size) { mutableStateOf(animals.size.coerceAtLeast(1).toString()) }
    var taskNotes by remember { mutableStateOf("") }
    var message by remember { mutableStateOf("") }
    var saving by remember { mutableStateOf(false) }
    var vaccineAdminDate by remember { mutableStateOf(today.toString()) }
    var vaccineProduct by remember { mutableStateOf("") }
    var vaccineDose by remember { mutableStateOf("") }
    var vaccineBatch by remember { mutableStateOf("") }
    var vaccineNextDue by remember { mutableStateOf("") }
    var vaccineVet by remember { mutableStateOf("") }
    var vaccineNotes by remember { mutableStateOf("") }
    var careStatus by remember { mutableStateOf("طبيعي") }
    var careNotes by remember { mutableStateOf("") }
    var campaignName by remember { mutableStateOf("") }
    var campaignRef by remember { mutableStateOf("") }
    var campaignTarget by remember { mutableStateOf("كل القطيع") }
    var campaignEnd by remember { mutableStateOf("") }
    var campaignStatus by remember { mutableStateOf("معلنة") }

    val selectedAnimals = when (feedGroup) {
        "نعاج" -> animals.filter { it.category.contains("نعجة") || it.category.contains("نعاج") }
        "فحول" -> animals.filter { it.category.contains("فحل") }
        "حملان" -> animals.filter { it.category.contains("حمل") || it.category.contains("خروف") }
        else -> animals
    }
    val daily = (dailyPerHead.replace(',', '.').toDoubleOrNull() ?: 0.0) * selectedAnimals.size
    val totalFeedCost = purchases.sumOf { it.totalPriceDzd ?: 0.0 }
    fun amountFrom(note: String): Double {
        val match = Regex("(?:المبلغ|قيمة البيع|قيمة المصروف|الوزن)[:： ]*([0-9]+(?:[.,][0-9]+)?)")
            .find(note) ?: return 0.0
        return match.groupValues[1].replace(',', '.').toDoubleOrNull() ?: 0.0
    }
    val sales = events.filter { it.eventType == "بيع" }.sumOf { amountFrom(it.notes) }
    val extraExpenses = events.filter { it.eventType == "مصروف" }.sumOf { amountFrom(it.notes) }
    val estimatedBalance = sales - totalFeedCost - extraExpenses
    val upcoming = tasks.mapNotNull { task ->
        val date = runCatching { LocalDate.parse(task.taskDate) }.getOrNull() ?: return@mapNotNull null
        val days = ChronoUnit.DAYS.between(today, date)
        if (!task.completed && days <= 7) task to days else null
    }.sortedBy { it.second }

    fun csv(value: String) = "\"" + value.replace("\"", "\"\"").replace("\n", " ") + "\""
    val exportText = buildString {
        appendLine("النوع,المعرف,التاريخ,التفاصيل")
        animals.forEach { appendLine("حيوان,${csv(it.tag)},,${csv("${it.sex} - ${it.category} - ${it.weightKg} كغ")}") }
        purchases.forEach { appendLine("شراء علف,${csv(it.feedName)},${csv(it.purchaseDate)},${csv("${it.quantityKg} كغ - ${it.totalPriceDzd ?: 0.0} دج")}") }
        consumptions.forEach { appendLine("استهلاك,${csv(it.feedName)},${csv(it.consumptionDate)},${csv("${it.quantityKg} كغ - ${it.animalCategory}")}") }
        tasks.forEach { appendLine("مهمة,${csv(it.taskType)},${csv(it.taskDate)},${csv(it.notes)}") }
        events.forEach { appendLine("حدث,${csv(it.animalTag)},${csv(it.eventDate)},${csv("${it.eventType} - ${it.notes}")}") }
    }

    Column(
        modifier = Modifier.fillMaxSize().verticalScroll(rememberScrollState()),
        verticalArrangement = Arrangement.spacedBy(10.dp)
    ) {
        Row(
            modifier = Modifier.fillMaxWidth(),
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.spacedBy(10.dp)
        ) {
            OutlinedButton(onClick = onBack) { Text("رجوع إلى الأقسام") }
            Text(section, style = MaterialTheme.typography.titleLarge, color = MaterialTheme.colorScheme.primary)
        }

        val vaccinePlan = remember(animals, events) {
            animals.flatMap { animal ->
                val birth = runCatching { LocalDate.parse(animal.birthDate) }.getOrNull()
                if (birth == null) emptyList() else listOf(
                    VaccinePlanItem(animal.tag, "clostridial-1", "التسمم المعوي والكزاز — الجرعة الأولى", birth.plusDays(45),
                        "توقيت استرشادي بعمر 6–8 أسابيع؛ يتغير حسب مناعة الأم ونشرة اللقاح."),
                    VaccinePlanItem(animal.tag, "clostridial-2", "التسمم المعوي والكزاز — الجرعة الثانية", birth.plusDays(69),
                        "بعد 3–4 أسابيع من الأولى، بشرط مطابقة الفاصل لنشرة المنتج."),
                    VaccinePlanItem(animal.tag, "ppr", "طاعون المجترات الصغيرة (PPR)", birth.plusDays(105),
                        "يُراجع عند عمر 3–4 أشهر؛ التنفيذ حسب برنامج المصالح البيطرية ونوع اللقاح وتداخل مناعة الأم."),
                    VaccinePlanItem(animal.tag, "pox-risk", "جدري الأغنام (عند وجود خطر)", birth.plusDays(120),
                        "لا يُعد موعداً روتينياً عاماً؛ لا يُعطى إلا بتوجيه المصالح البيطرية أو الطبيب.")
                )
            }
        }
        fun vaccineRecorded(item: VaccinePlanItem) =
            vaccineRecords.any { it.animalTag == item.animalTag && it.vaccineKey == item.key } ||
                events.any { it.animalTag == item.animalTag && it.eventType == "تطعيم" && it.notes.contains("VAX:" + item.key) }
        val careByTag = careStatuses.associateBy { it.animalTag }
        LaunchedEffect(vaccinePlan, events, vaccineRecords) {
            vaccinePlan.forEach { item ->
                val labels = listOf("تبقى 3 أيام — موعد استرشادي", "موعد مراجعة التطعيم اليوم الساعة 08:00")
                if (vaccineRecorded(item)) {
                    labels.forEach { cancelVaccineAlarm(context, item.animalTag, item.key, it) }
                } else {
                    scheduleVaccineAlarm(context, item.animalTag, item.key, item.title, item.dueDate, item.dueDate.minusDays(3), labels[0])
                    scheduleVaccineAlarm(context, item.animalTag, item.key, item.title, item.dueDate, item.dueDate, labels[1])
                }
            }
            val latest = vaccineRecords.groupBy { it.animalTag to it.vaccineKey }
                .values.mapNotNull { records -> records.maxByOrNull { it.administeredDate + "|" + it.id } }
            latest.forEach { record ->
                val key = record.vaccineKey + "-next"
                val labels = listOf("تبقى 3 أيام على الجرعة التالية", "موعد الجرعة التالية المسجل")
                val next = runCatching { LocalDate.parse(record.nextDueDate) }.getOrNull()
                if (next == null) {
                    labels.forEach { cancelVaccineAlarm(context, record.animalTag, key, it) }
                } else {
                    scheduleVaccineAlarm(context, record.animalTag, key, record.vaccineName, next, next.minusDays(3), labels[0])
                    scheduleVaccineAlarm(context, record.animalTag, key, record.vaccineName, next, next, labels[1])
                }
            }
        }
        when (section) {
            "برنامج التطعيمات" -> {
                Text("سجل التطعيمات المنظم لكل حيوان", style = MaterialTheme.typography.titleMedium)
                Text("المواعيد قوالب استرشادية وليست وصفة أو أمراً بإعطاء اللقاح. القرار للطبيب البيطري ونشرة المنتج والبرنامج الرسمي.", color = MaterialTheme.colorScheme.error, style = MaterialTheme.typography.bodySmall)
                val dueItems = vaccinePlan.map { item ->
                    val recorded = vaccineRecorded(item)
                    val review = careByTag[item.animalTag]?.status?.let { it != "طبيعي" } ?: false
                    val days = ChronoUnit.DAYS.between(today, item.dueDate)
                    val status = when { recorded -> "مكتمل"; review -> "مراجعة بيطرية"; days < 0 -> "متأخر"; days <= 7 -> if (days == 0L) "مستحق" else "قريب"; else -> "مجدول" }
                    Triple(item, status, days)
                }
                listOf("قريب", "مستحق", "متأخر", "مراجعة بيطرية", "مكتمل").forEach { label ->
                    AssistChip(onClick = {}, label = { Text("$label: ${dueItems.count { it.second == label }}") })

                                OutlinedTextField(vaccineProduct, { vaccineProduct = it }, label = { Text("اسم اللقاح / المنتج") }, modifier = Modifier.fillMaxWidth())
                                OutlinedTextField(vaccineDose, { vaccineDose = it }, label = { Text("الجرعة والكمية حسب النشرة") }, modifier = Modifier.fillMaxWidth())
                                OutlinedTextField(vaccineBatch, { vaccineBatch = it }, label = { Text("رقم الدفعة Lot") }, modifier = Modifier.fillMaxWidth())
                                OutlinedTextField(vaccineNextDue, { vaccineNextDue = it }, label = { Text("موعد الجرعة التالية اختياري YYYY-MM-DD") }, modifier = Modifier.fillMaxWidth())
                                OutlinedTextField(vaccineVet, { vaccineVet = it }, label = { Text("الطبيب البيطري") }, modifier = Modifier.fillMaxWidth())
                                OutlinedTextField(vaccineNotes, { vaccineNotes = it }, label = { Text("ملاحظات") }, modifier = Modifier.fillMaxWidth())
                                Button(enabled = !saving && vaccineProduct.isNotBlank() && vaccineDose.isNotBlank() && vaccineBatch.isNotBlank(), onClick = {
                                    val next = vaccineNextDue.trim()
                                    if (runCatching { LocalDate.parse(vaccineAdminDate) }.isFailure || (next.isNotEmpty() && runCatching { LocalDate.parse(next) }.isFailure)) message = "تحقق من تاريخ إعطاء الجرعة والجرعة التالية."
                                    else {
                                        saving = true
                                        onAddVaccineRecord(VaccineDoseRecord(animalTag=item.animalTag, vaccineKey=item.key, vaccineName=item.title, productName=vaccineProduct.trim(), administeredDate=vaccineAdminDate, doseText=vaccineDose.trim(), batchNumber=vaccineBatch.trim(), nextDueDate=next, veterinarian=vaccineVet.trim(), notes=vaccineNotes.trim())) { ok ->
                                            saving = false
                                            message = if (ok) "تم حفظ الجرعة في السجل." else "تعذر حفظ الجرعة."
                                            if (ok) { vaccineAdminDate=today.toString(); vaccineProduct=""; vaccineDose=""; vaccineBatch=""; vaccineNextDue=""; vaccineVet=""; vaccineNotes="" }
                                        }
                                    }
                                }, modifier = Modifier.fillMaxWidth()) { Text("تأكيد إعطاء اللقاح وحفظ الجرعة") }
                            }
                        }
                    }
                }
                Text("حالة الحيوان قبل التطعيم", style = MaterialTheme.typography.titleMedium)
                DropdownField("الحيوان", targetTag, animals.map { it.tag }, { targetTag = it })

                }, modifier = Modifier.fillMaxWidth()) { Text("حفظ حالة الحيوان") }
                careStatuses.forEach { c -> ListItem(headlineContent={Text(c.animalTag + " • " + c.status)}, supportingContent={Text(c.notes)}) }
                Text("لقاحات مرتبطة بالخطر أو القرار الرسمي — لا تُجدول تلقائياً", style = MaterialTheme.typography.titleSmall)
                listOf("PPR وجدري الأغنام: وفق الوضع الوبائي وتوجيه المصالح البيطرية.", "الحمى القلاعية والبروسيلا: وفق البرنامج الرسمي والقيود التنظيمية.", "الباستوريلا والجمرة الخبيثة والحمى الزرقاء: حسب تقييم الخطر والطبيب.", "لا يحدد التطبيق الجرعات أو طرق الحقن؛ اتبع النشرة المعتمدة والطبيب.").forEach { Text("• " + it, style = MaterialTheme.typography.bodySmall) }
                if (message.isNotBlank()) Text(message)
            }
            "حملات التلقيح الجزائرية" -> {
                Text("سجل الحملات والبلاغات البيطرية الجزائرية", style = MaterialTheme.typography.titleMedium)
                Text("إدخال يدوي من البلاغ الرسمي؛ لا يوجد اتصال حكومي آلي. طابق الولاية والبلدية والفئة المستهدفة قبل إشراك الحيوانات.", style = MaterialTheme.typography.bodySmall)
                OutlinedTextField(campaignName, { campaignName=it }, label={Text("اسم الحملة / المرض")}, modifier=Modifier.fillMaxWidth())
                OutlinedTextField(campaignRef, { campaignRef=it }, label={Text("الجهة والمرجع أو رابط الإعلان")}, modifier=Modifier.fillMaxWidth())
                OutlinedTextField(campaignTarget, { campaignTarget=it }, label={Text("الفئة المستهدفة / الولاية والبلدية")}, modifier=Modifier.fillMaxWidth())
                CalendarDateField(label="بداية الحملة", value=eventDate, onDateSelected={eventDate=it})
                OutlinedTextField(campaignEnd, { campaignEnd=it }, label={Text("نهاية الحملة اختيارية YYYY-MM-DD")}, modifier=Modifier.fillMaxWidth())
                DropdownField("حالة الحملة", campaignStatus, listOf("معلنة","جارية","منتهية","ملغاة"), {campaignStatus=it})
                OutlinedTextField(eventNotes, {eventNotes=it}, label={Text("الشروط والملاحظات")}, modifier=Modifier.fillMaxWidth())
                Button(enabled=!saving && campaignName.isNotBlank() && campaignRef.isNotBlank(), onClick={
                    val st=runCatching { LocalDate.parse(eventDate) }.getOrNull()
                    val en=if(campaignEnd.isBlank()) null else runCatching { LocalDate.parse(campaignEnd) }.getOrNull()
                    if(st==null || (campaignEnd.isNotBlank() && en==null) || (en!=null && en.isBefore(st))) message="تحقق من تاريخ بداية ونهاية الحملة."
                    else {
                        saving=true
                        onAddVaccineCampaign(VaccineCampaign(name=campaignName.trim(), authorityReference=campaignRef.trim(), targetGroup=campaignTarget.trim(), startDate=eventDate, endDate=campaignEnd.trim(), status=campaignStatus, notes=eventNotes.trim())) { ok ->
                            saving=false
                            message=if(ok) "تم حفظ الحملة ومرجعها." else "تعذر حفظ الحملة."
                            if(ok) {campaignName="";campaignRef="";eventNotes=""}
                        }
                    }
                }, modifier=Modifier.fillMaxWidth()) { Text("حفظ الحملة") }
                vaccineCampaigns.forEach { c -> ListItem(headlineContent={Text(c.name+" • "+c.status)}, supportingContent={Text("${c.startDate} – ${c.endDate} • ${c.targetGroup} • ${c.authorityReference}")}) }
                if(message.isNotBlank()) Text(message)
            }
            "التغذية الذكية" -> {
                Text("حاسبة الاحتياج اليومي", style = MaterialTheme.typography.titleMedium)
                DropdownField("الفئة", feedGroup, listOf("القطيع كله", "نعاج", "فحول", "حملان"), { feedGroup = it })
                OutlinedTextField(
                    value = dailyPerHead, onValueChange = { dailyPerHead = it },
                    label = { Text("العلف اليومي للحيوان الواحد بالكغ") },
                    supportingText = { Text("أدخل الحصة التي تعتمدها أنت أو يحددها المختص البيطري.") },
                    modifier = Modifier.fillMaxWidth()
                )
                Card {
                    Column(Modifier.padding(14.dp), verticalArrangement = Arrangement.spacedBy(5.dp)) {
                        Text("عدد الحيوانات: ${selectedAnimals.size}")
                        Text("الاحتياج اليومي: ${"%.2f".format(Locale.US, daily)} كغ")
                        Text("7 أيام: ${"%.1f".format(Locale.US, daily * 7)} كغ")
                        Text("30 يوماً: ${"%.1f".format(Locale.US, daily * 30)} كغ")
                        Text("90 يوماً: ${"%.1f".format(Locale.US, daily * 90)} كغ")
                    }
                }
                Text("مخزون الأعلاف", style = MaterialTheme.typography.titleMedium)
                val feedNames = (purchases.map { it.feedName } + consumptions.map { it.feedName }).distinct().sorted()
                if (feedNames.isEmpty()) Text("لم تسجل مشتريات أو استهلاكاً بعد.")
                feedNames.forEach { name ->
                    val bought = purchases.filter { it.feedName == name }.sumOf { it.quantityKg }
                    val used = consumptions.filter { it.feedName == name }.sumOf { it.quantityKg }
                    ListItem(
                        headlineContent = { Text(name) },
                        supportingContent = { Text("مشتَرى: ${"%.1f".format(Locale.US, bought)} كغ • مستهلك: ${"%.1f".format(Locale.US, used)} كغ") },
                        trailingContent = { Text("${"%.1f".format(Locale.US, bought - used)} كغ",
                            color = if (bought - used <= 0) MaterialTheme.colorScheme.error else MaterialTheme.colorScheme.primary) }
                    )
                    HorizontalDivider()
                }
            }
            "الصحة والتناسل" -> {
                Text("سجل صحي وتناسلي موحد", style = MaterialTheme.typography.titleMedium)
                DropdownField("الحيوان", targetTag, listOf("عام") + animals.map { it.tag }, { targetTag = it })
                DropdownField("نوع الحدث", eventType,
                    listOf("تلقيح", "تطعيم", "علاج", "فحص بيطري", "ولادة", "فطام", "بيع", "مصروف"),
                    { eventType = it })
                CalendarDateField(
                    label = "تاريخ الحدث",
                    value = eventDate,
                    onDateSelected = { eventDate = it }
                )
                if (eventType == "بيع" || eventType == "مصروف") {
                    OutlinedTextField(eventAmount, { eventAmount = it }, label = { Text("المبلغ بالدينار الجزائري") }, modifier = Modifier.fillMaxWidth())
                }
                OutlinedTextField(eventNotes, { eventNotes = it }, label = { Text("التفاصيل والملاحظات") }, modifier = Modifier.fillMaxWidth())
                Button(
                    enabled = !saving,
                    onClick = {
                        val amount = eventAmount.replace(',', '.').toDoubleOrNull()
                        if (runCatching { LocalDate.parse(eventDate) }.isFailure ||
                            ((eventType == "بيع" || eventType == "مصروف") && (amount == null || amount <= 0.0))
                        ) {
                            message = "تحقق من التاريخ والمبلغ."
                        } else {
                            val detail = buildString {
                                if (eventType == "بيع") append("قيمة البيع: $amount دج. ")
                                if (eventType == "مصروف") append("قيمة المصروف: $amount دج. ")
                                append(eventNotes.trim())
                            }
                            saving = true
                            onAddEvent(targetTag, eventType, eventDate, detail) { ok ->
                                saving = false
                                message = if (ok) "تم حفظ السجل." else "تعذر حفظ السجل."
                                if (ok) { eventAmount = ""; eventNotes = "" }
                            }
                        }
                    },
                    modifier = Modifier.fillMaxWidth()
                ) { Text(if (saving) "جارٍ الحفظ..." else "حفظ الحدث") }
                if (message.isNotBlank()) Text(message)
                Text("آخر السجلات", style = MaterialTheme.typography.titleMedium)
                events.take(8).forEach { event ->
                    ListItem(
                        headlineContent = { Text("${event.eventType} • ${event.animalTag}") },
                        supportingContent = { Text("${event.eventDate} — ${event.notes}") }
                    )
                    HorizontalDivider()
                }
            }
            "الوزن والنمو" -> {
                Text("تسجيل متابعة الوزن", style = MaterialTheme.typography.titleMedium)
                DropdownField("الحيوان", targetTag, animals.map { it.tag }, { targetTag = it })
                CalendarDateField(
                    label = "تاريخ القياس",
                    value = eventDate,
                    onDateSelected = { eventDate = it }
                )
                OutlinedTextField(eventAmount, { eventAmount = it }, label = { Text("الوزن بالكغ") }, modifier = Modifier.fillMaxWidth())
                Button(
                    enabled = !saving && animals.isNotEmpty(),
                    onClick = {
                        val weight = eventAmount.replace(',', '.').toDoubleOrNull()
                        if (weight == null || weight <= 0 || runCatching { LocalDate.parse(eventDate) }.isFailure) {
                            message = "أدخل وزناً وتاريخاً صحيحين."
                        } else {
                            saving = true
                            onAddEvent(targetTag, "وزن", eventDate, "الوزن: $weight كغ") { ok ->
                                saving = false
                                message = if (ok) "تم تسجيل الوزن." else "تعذر تسجيل الوزن."
                                if (ok) eventAmount = ""
                            }
                        }
                    }, modifier = Modifier.fillMaxWidth()
                ) { Text("حفظ قياس الوزن") }
                if (message.isNotBlank()) Text(message)
                Text("سجل الأوزان", style = MaterialTheme.typography.titleMedium)
                events.filter { it.eventType == "وزن" && it.animalTag == targetTag }
                    .sortedByDescending { it.eventDate }.take(12).forEach { item ->
                        ListItem(
                            headlineContent = { Text("${amountFrom(item.notes)} كغ") },
                            supportingContent = { Text(item.eventDate) }
                        )
                    }
            }
            "المواعيد والتنبيهات" -> {
                Text("إضافة موعد أو مهمة", style = MaterialTheme.typography.titleMedium)
                DropdownField("نوع الموعد", taskType,
                    listOf("موعد تلقيح", "ولادة متوقعة", "تطعيم", "علاج", "فحص بيطري", "قص الصوف", "شراء علف", "أخرى"),
                    { taskType = it })
                CalendarDateField(
                    label = "تاريخ الموعد",
                    value = taskDate,
                    onDateSelected = { taskDate = it }
                )
                OutlinedTextField(taskCount, { taskCount = it }, label = { Text("عدد الحيوانات") }, modifier = Modifier.fillMaxWidth())
                OutlinedTextField(taskNotes, { taskNotes = it }, label = { Text("ملاحظات") }, modifier = Modifier.fillMaxWidth())
                Button(
                    enabled = !saving,
                    onClick = {
                        val count = taskCount.toIntOrNull()
                        if (count == null || count <= 0 || runCatching { LocalDate.parse(taskDate) }.isFailure) {
                            message = "تحقق من التاريخ وعدد الحيوانات."
                        } else {
                            saving = true
                            onAddTask(taskType, taskDate, count, taskNotes) { ok ->
                                saving = false
                                message = if (ok) "تمت إضافة الموعد." else "تعذرت إضافة الموعد."
                                if (ok) taskNotes = ""
                            }
                        }
                    }, modifier = Modifier.fillMaxWidth()
                ) { Text("إضافة موعد") }
                if (message.isNotBlank()) Text(message)
                Text("المواعيد المتأخرة والقريبة خلال 7 أيام", style = MaterialTheme.typography.titleMedium)
                if (upcoming.isEmpty()) Text("لا توجد مواعيد متأخرة أو قريبة.")
                upcoming.forEach { (task, days) ->
                    ListItem(
                        headlineContent = { Text(task.taskType) },
                        supportingContent = { Text("${task.taskDate} • ${task.animalCount} حيوان • ${task.notes}") },
                        trailingContent = {
                            Text(if (days < 0) "متأخر ${-days} يوم" else if (days == 0L) "اليوم" else "بعد ${days} يوم",
                                color = if (days < 0) MaterialTheme.colorScheme.error else MaterialTheme.colorScheme.primary)
                        }
                    )
                    HorizontalDivider()
                }
            }
            else -> {
                Text("ملخص مالي تقديري", style = MaterialTheme.typography.titleMedium)
                SummaryCard("تكلفة مشتريات الأعلاف المسجلة", "${"%.0f".format(Locale.US, totalFeedCost)} دج")
                SummaryCard("المصاريف الإضافية المسجلة", "${"%.0f".format(Locale.US, extraExpenses)} دج")
                SummaryCard("إيرادات البيع المسجلة", "${"%.0f".format(Locale.US, sales)} دج")
                SummaryCard("الرصيد التقديري", "${"%.0f".format(Locale.US, estimatedBalance)} دج")
                Text("هذا ملخص للسجلات المدخلة فقط، وليس ربحاً محاسبياً كاملاً.", style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant)
                Button(
                    onClick = {
                        val intent = Intent(Intent.ACTION_SEND).apply {
                            type = "text/csv"
                            putExtra(Intent.EXTRA_SUBJECT, "نسخة بيانات مواشي")
                            putExtra(Intent.EXTRA_TEXT, exportText)
                        }
                        context.startActivity(Intent.createChooser(intent, "تصدير بيانات المزرعة"))
                    }, modifier = Modifier.fillMaxWidth()
                ) { Text("تصدير سجلات المزرعة CSV") }
                Text("يشمل التصدير الحيوانات والمشتريات والاستهلاك والمهام والأحداث المسجلة.", style = MaterialTheme.typography.bodySmall)
            }
        }
    }
