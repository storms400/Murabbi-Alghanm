class FarmViewModel(private val dao: FarmDao) : ViewModel() {

    val animals = dao.activeAnimals().stateIn(
        viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList()
    )

    val purchases = dao.purchases().stateIn(
        viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList()
    )

    val consumptions = dao.consumptions().stateIn(
        viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList()
    )

    val tasks = dao.tasks().stateIn(
        viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList()
    )

    val events = dao.events().stateIn(
        viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList()
    )

    fun addAnimal(
        tag: String,
        sex: String,
        category: String,
        weight: Double,
        onResult: (String?) -> Unit
    ) {
        viewModelScope.launch {
            try {
                dao.insertAnimal(
                    Animal(
                        tag = tag.trim(),
                        sex = sex,
                        category = category,
                        weightKg = weight
                    )
                )
                onResult(null)
            } catch (_: Exception) {
                onResult("تعذر الحفظ: تأكد أن رمز الحيوان غير مكرر.")
            }
        }
    }

    fun archive(animal: Animal, onResult: (Boolean) -> Unit) {
        viewModelScope.launch {
            try {
                dao.archiveAnimal(animal.tag, "أرشفة يدوية")
                onResult(true)
            } catch (_: Exception) {
                onResult(false)
            }
        }
    }

    fun updateAnimal(
        animal: Animal,
        onResult: (Boolean) -> Unit
    ) {
        viewModelScope.launch {
            try {
                dao.updateAnimal(animal)
                onResult(true)
            } catch (_: Exception) {
                onResult(false)
            }
        }
    }

    fun addAnimalEvent(
        tag: String,
        type: String,
        date: String,
        notes: String,
        onResult: (Boolean) -> Unit
    ) {
        viewModelScope.launch {
            try {
                dao.addEvent(
                    HerdEvent(
                        animalTag = tag,
                        eventType = type,
                        eventDate = date,
                        notes = notes
                    )
                )
                onResult(true)
            } catch (_: Exception) {
                onResult(false)
            }
        }
    }

    fun addPurchase(
        name: String,
        qty: Double,
        price: Double?,
        supplier: String,
        onResult: (Boolean) -> Unit
    ) {
        viewModelScope.launch {
            try {
                dao.addPurchase(
                    FeedPurchase(
                        feedName = name.trim(),
                        quantityKg = qty,
                        totalPriceDzd = price,
                        supplier = supplier,
                        purchaseDate = LocalDate.now().toString()
                    )
                )
                onResult(true)
            } catch (_: Exception) {
                onResult(false)
            }
        }
    }

    fun addConsumption(
        name: String,
        qty: Double,
        category: String,
        count: Int,
        notes: String,
        onResult: (Boolean) -> Unit
    ) {
        viewModelScope.launch {
            try {
                val saved = dao.addConsumptionSafely(
                    FeedConsumption(
                        feedName = name,
                        quantityKg = qty,
                        consumptionDate = LocalDate.now().toString(),
                        animalCategory = category,
                        animalCount = count,
                        notes = notes
                    )
                )
                onResult(saved)
            } catch (_: Exception) {
                onResult(false)
            }
        }
    }

    fun addTask(
        type: String,
        date: String,
        count: Int,
        notes: String,
        onResult: (Boolean) -> Unit
    ) {
        viewModelScope.launch {
            try {
                dao.addTask(
                    FarmTask(
                        taskType = type,
                        taskDate = date,
                        animalCount = count,
                        notes = notes
                    )
                )
                onResult(true)
            } catch (_: Exception) {
                onResult(false)
            }
        }
    }
}

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun FarmScreen(vm: FarmViewModel) {

    val animals by vm.animals.collectAsState()
    val purchases by vm.purchases.collectAsState()
    val consumptions by vm.consumptions.collectAsState()
    val tasks by vm.tasks.collectAsState()
    val events by vm.events.collectAsState()

    var selectedTab by remember { mutableIntStateOf(0) }
    var selectedAnimal by remember { mutableStateOf<Animal?>(null) }
    var screenMessage by remember { mutableStateOf("") }

    val tabs = listOf("الرئيسية", "القطيع", "المشتريات", "التغذية", "الرعاية")

    Scaffold(
        topBar = {
            TopAppBar(title = { Text("مربي الغنم 🐑") })
        },
        bottomBar = {
            NavigationBar {
                tabs.forEachIndexed { index, title ->
                    NavigationBarItem(
                        selected = selectedTab == index,
                        onClick = {
                            selectedTab = index
                            selectedAnimal = null
                            screenMessage = ""
                        },
                        icon = {
                            Icon(
                                imageVector = when (index) {
                                    0 -> androidx.compose.material.icons.Icons.Default.Home
                                    1 -> androidx.compose.material.icons.Icons.Default.Pets
                                    2 -> androidx.compose.material.icons.Icons.Default.ShoppingCart
                                    3 -> androidx.compose.material.icons.Icons.Default.Inventory
                                    else -> androidx.compose.material.icons.Icons.Default.CalendarMonth
                                },
                                contentDescription = title
                            )
                        },
                        label = { Text(title) }
                    )
                }
            }
        }
    ) { padding ->

        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(padding)
                .padding(12.dp)
        ) {

            if (selectedAnimal != null) {
                val animal = selectedAnimal!!

                AnimalDetailsScreen(
                    animal = animal,
                    events = events.filter { it.animalTag == animal.tag },
                    onBack = { selectedAnimal = null },
                    onUpdate = { updated ->
                        vm.updateAnimal(updated) { success ->
                            screenMessage = if (success)
                                "تم تحديث البيانات بنجاح."
                            else
                                "تعذر تحديث البيانات."
                        }
                    },
                    onAddEvent = { type, date, notes ->
                        vm.addAnimalEvent(
                            animal.tag, type, date, notes
                        ) { success ->
                            screenMessage = if (success)
                                "تم تسجيل الحدث."
                            else
                                "تعذر تسجيل الحدث."
                        }
                    },
                    onArchive = {
                        vm.archive(animal) { success ->
                            if (success) selectedAnimal = null
                            else screenMessage = "تعذرت أرشفة الحيوان."
                        }
                    }
                )

                if (screenMessage.isNotBlank()) {
                    Text(screenMessage)
                }

            } else {

                when (selectedTab) {

                    0 -> HomeScreen(
                        animals, purchases, consumptions, tasks, events,
                        onAddEvent = { tag, type, date, notes, result ->
                            vm.addAnimalEvent(tag, type, date, notes, result)
                        },
                        onAddTask = { type, date, count, notes, result ->
                            vm.addTask(type, date, count, notes, result)
                        }
                    )

                    1 -> HerdScreen(
                        animals = animals,
                        onOpen = { selectedAnimal = it },
                        onAdd = { tag, sex, category, weight, result ->
                            vm.addAnimal(
                                tag, sex, category, weight, result
                            )
                        }
                    )

                    2 -> PurchasesScreen(
                        purchases = purchases,
                        onAdd = { name, qty, price, supplier, result ->
                            vm.addPurchase(
                                name, qty, price, supplier, result
                            )
                        }
                    )

                    3 -> FeedingScreen(
                        purchases = purchases,
                        consumptions = consumptions,
                        animals = animals,
                        onAdd = { name, qty, category, count, notes, result ->
                            vm.addConsumption(
                                name, qty, category, count, notes, result
                            )
                        }
                    )

                    else -> CareScreen(
                        tasks = tasks,
                        onAdd = { type, date, count, notes, result ->
                            vm.addTask(type, date, count, notes, result)
                        }
                    )
                }
            }
        }
    }
}

@Composable
fun HomeScreen(
    animals: List<Animal>,
    purchases: List<FeedPurchase>,
    consumptions: List<FeedConsumption>,
    tasks: List<FarmTask>,
    events: List<HerdEvent>,
    onAddEvent: (String, String, String, String, (Boolean) -> Unit) -> Unit,
    onAddTask: (String, String, Int, String, (Boolean) -> Unit) -> Unit
) {
    val totalPurchased = purchases.sumOf { it.quantityKg }
    val totalConsumed = consumptions.sumOf { it.quantityKg }
    val stock = (totalPurchased - totalConsumed).coerceAtLeast(0.0)
    val formatter = java.text.DecimalFormat("#,##0.##")

    LazyColumn(
        modifier = Modifier.fillMaxSize(),
        verticalArrangement = Arrangement.spacedBy(14.dp)
    ) {
        item {
            Card(
                modifier = Modifier.fillMaxWidth(),
                colors = CardDefaults.cardColors(
                    containerColor = MaterialTheme.colorScheme.primary
                ),
                shape = MaterialTheme.shapes.extraLarge
            ) {
                Column(Modifier.padding(20.dp), verticalArrangement = Arrangement.spacedBy(6.dp)) {
                    Text("مرحباً بك في مزرعتك", color = androidx.compose.ui.graphics.Color.White,
                        style = MaterialTheme.typography.titleMedium)
                    Text("مربي الغنم", color = androidx.compose.ui.graphics.Color.White,
                        style = MaterialTheme.typography.headlineMedium)
                    Text("تابع القطيع والأعلاف من مكان واحد",
                        color = androidx.compose.ui.graphics.Color.White.copy(alpha = 0.88f))
                }
            }
        }
        item {
            Text("ملخص المزرعة", style = MaterialTheme.typography.titleLarge,
                color = MaterialTheme.colorScheme.primary)
        }
        item {
            Row(horizontalArrangement = Arrangement.spacedBy(10.dp)) {
                DashboardMetric("القطيع النشط", animals.size.toString(), "🐑", Modifier.weight(1f))
                DashboardMetric("مخزون العلف", "${formatter.format(stock)} كغ", "🌾", Modifier.weight(1f))
            }
        }
        item {
            Row(horizontalArrangement = Arrangement.spacedBy(10.dp)) {
                DashboardMetric("المشتريات", purchases.size.toString(), "🛒", Modifier.weight(1f))
                DashboardMetric("المهام", tasks.size.toString(), "📋", Modifier.weight(1f))
            }
        }
        item {
            Card(
                modifier = Modifier.fillMaxWidth(),
                colors = CardDefaults.cardColors(containerColor = MaterialTheme.colorScheme.surface),
                shape = MaterialTheme.shapes.large
            ) {
                Column(Modifier.padding(16.dp), verticalArrangement = Arrangement.spacedBy(8.dp)) {
                    Text("حركة الأعلاف", style = MaterialTheme.typography.titleMedium)
                    Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.SpaceBetween) {
                        Text("إجمالي المشتريات")
                        Text("${formatter.format(totalPurchased)} كغ",
                            color = MaterialTheme.colorScheme.primary)
                    }
                    HorizontalDivider()
                    Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.SpaceBetween) {
                        Text("إجمالي الاستهلاك")
                        Text("${formatter.format(totalConsumed)} كغ",
                            color = MaterialTheme.colorScheme.secondary)
                    }
                }
            }
        }
        item {
            SmartFarmPanel(
                animals = animals,
                purchases = purchases,
                consumptions = consumptions,
                tasks = tasks,
                events = events,
                onAddEvent = onAddEvent,
                onAddTask = onAddTask
            )
        }
        item {
            Card(
                modifier = Modifier.fillMaxWidth(),
                colors = CardDefaults.cardColors(containerColor = MaterialTheme.colorScheme.secondaryContainer),
                shape = MaterialTheme.shapes.large
            ) {
                Column(Modifier.padding(16.dp)) {
                    Text("نصيحة", style = MaterialTheme.typography.titleMedium,
                        color = MaterialTheme.colorScheme.primary)
                    Text("سجّل المشتريات والاستهلاك بانتظام لمتابعة رصيد الأعلاف بدقة.")
                }
            }
        }
    }
}

@Composable
fun HerdScreen(
    animals: List<Animal>,
    onOpen: (Animal) -> Unit,
    onAdd: (String, String, String, Double, (String?) -> Unit) -> Unit
) {
    var tag by remember { mutableStateOf("") }
    var sex by remember { mutableStateOf("أنثى") }
    var category by remember { mutableStateOf("نعجة") }
    var weight by remember { mutableStateOf("") }
    var error by remember { mutableStateOf("") }
    var saving by remember { mutableStateOf(false) }

    Column(modifier = Modifier.fillMaxSize()) {

        Text("القطيع", style = MaterialTheme.typography.headlineSmall)

        OutlinedTextField(
            value = tag,
            onValueChange = { tag = it },
            label = { Text("رقم أو رمز الحيوان") },
            modifier = Modifier.fillMaxWidth(),
            enabled = !saving
        )

        OutlinedTextField(
            value = weight,
            onValueChange = { weight = it },
            label = { Text("الوزن بالكغ") },
            modifier = Modifier.fillMaxWidth(),
            enabled = !saving
        )

        Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
            DropdownField(
                label = "الجنس",
                value = sex,
                options = listOf("أنثى", "ذكر"),
                onSelect = { sex = it },
                modifier = Modifier.weight(1f)
            )
            DropdownField(
                label = "الفئة",
                value = category,
                options = listOf("نعجة", "فحل", "حمل", "خروف"),
                onSelect = { category = it },
                modifier = Modifier.weight(1f)
            )
        }

        Button(
            enabled = !saving,
            onClick = {
                val w = weight.toDoubleOrNull()
                if (tag.isBlank() || w == null || w <= 0 ||
                    w.isNaN() || w.isInfinite()
                ) {
                    error = "أدخل رمز الحيوان ووزناً صحيحاً."
                } else {
                    saving = true
                    error = ""
                    onAdd(tag, sex, category, w) { result ->
                        saving = false
                        if (result == null) {
                            tag = ""
                            weight = ""
                            error = "تمت إضافة الحيوان."
                        } else {
                            error = result
                        }
                    }
                }
            },
            modifier = Modifier.fillMaxWidth()
        ) {
            Text(if (saving) "جارٍ الحفظ..." else "إضافة حيوان")
        }

        if (error.isNotBlank()) {
            Text(error, color = MaterialTheme.colorScheme.error)
        }

        HorizontalDivider()
        Text("الحيوانات النشطة: ${animals.size}")

        LazyColumn(
            modifier = Modifier.weight(1f),
            verticalArrangement = Arrangement.spacedBy(6.dp)
        ) {
            items(animals, key = { it.tag }) { animal ->
                Card(
                    onClick = { onOpen(animal) },
                    modifier = Modifier.fillMaxWidth()
                ) {
                    Column(Modifier.padding(12.dp)) {
                        Text("الرمز: ${animal.tag}")
                        Text("${animal.category} - ${animal.sex}")
                        Text("الوزن: ${animal.weightKg} كغ")
                    }
                }
            }
        }
    }
}

@Composable
fun PurchasesScreen(
    purchases: List<FeedPurchase>,
    onAdd: (String, Double, Double?, String, (Boolean) -> Unit) -> Unit
) {
    val feeds = listOf(
        "الشعير", "التبن", "النخالة",
        "الذرة", "البرسيم الأخضر", "البرسيم المجفف"
    )

    var feed by remember { mutableStateOf(feeds[0]) }
    var quantity by remember { mutableStateOf("") }
    var price by remember { mutableStateOf("") }
    var supplier by remember { mutableStateOf("") }
    var message by remember { mutableStateOf("") }
    var saving by remember { mutableStateOf(false) }

    Column(Modifier.fillMaxSize()) {

        Text("مشتريات الأعلاف", style = MaterialTheme.typography.headlineSmall)

        DropdownField("نوع العلف", feed, feeds, { feed = it })

        OutlinedTextField(
            quantity, { quantity = it },
            label = { Text("الكمية بالكغ") },
            modifier = Modifier.fillMaxWidth(),
            enabled = !saving
        )

        OutlinedTextField(
            price, { price = it },
            label = { Text("السعر الإجمالي بالدينار (اختياري)") },
            modifier = Modifier.fillMaxWidth(),
            enabled = !saving
        )

        OutlinedTextField(
            supplier, { supplier = it },
            label = { Text("المورد (اختياري)") },
            modifier = Modifier.fillMaxWidth(),
            enabled = !saving
        )

        Button(
            enabled = !saving,
            onClick = {
                val q = quantity.toDoubleOrNull()
                val p = if (price.isBlank()) null else price.toDoubleOrNull()

                if (q == null || q <= 0 || q.isNaN() || q.isInfinite() ||
                    (price.isNotBlank() &&
                        (p == null || p < 0 || p.isNaN() || p.isInfinite()))
                ) {
                    message = "تحقق من الكمية والسعر."
                } else {
                    saving = true
                    onAdd(feed, q, p, supplier) { success ->
                        saving = false
                        if (success) {
                            quantity = ""
                            price = ""
                            supplier = ""
                            message = "تم تسجيل الشراء بنجاح."
                        } else {
                            message = "تعذر تسجيل الشراء."
                        }
                    }
                }
            },
            modifier = Modifier.fillMaxWidth()
        ) {
            Text(if (saving) "جارٍ الحفظ..." else "تسجيل الشراء")
        }

        if (message.isNotBlank()) Text(message)

        HorizontalDivider()
        Text("سجل المشتريات", style = MaterialTheme.typography.titleMedium)

        LazyColumn(Modifier.weight(1f)) {
            items(purchases, key = { it.id }) { purchase ->
                ListItem(
                    headlineContent = { Text(purchase.feedName) },
                    supportingContent = {
                        Text("${purchase.quantityKg} كغ - ${purchase.purchaseDate}")
                    },
                    trailingContent = {
                        Text(
                            purchase.totalPriceDzd?.let { "$it دج" }
                                ?: "السعر غير محدد"
                        )
                    }
                )
                HorizontalDivider()
            }
        }
    }
}

@Composable
fun FeedingScreen(
    purchases: List<FeedPurchase>,
    consumptions: List<FeedConsumption>,
    animals: List<Animal>,
    onAdd: (String, Double, String, Int, String, (Boolean) -> Unit) -> Unit
) {
    val feeds = listOf(
        "الشعير", "التبن", "النخالة",
        "الذرة", "البرسيم الأخضر", "البرسيم المجفف"
    )

    val categories = listOf(
        "القطيع كله", "نعاج", "حوامل",
        "مرضعات", "فحول", "حملان"
    )

    var feed by remember { mutableStateOf(feeds[0]) }
    var category by remember { mutableStateOf(categories[0]) }
    var quantity by remember { mutableStateOf("") }
    var count by remember(animals.size) {
        mutableStateOf(animals.size.toString())
    }
    var notes by remember { mutableStateOf("") }
    var message by remember { mutableStateOf("") }
    var saving by remember { mutableStateOf(false) }

    val balances = feeds.associateWith { name ->
        purchases.filter { it.feedName == name }.sumOf { it.quantityKg } -
            consumptions.filter { it.feedName == name }.sumOf { it.quantityKg }
    }

    Column(Modifier.fillMaxSize()) {

        Text("التغذية والمخزون", style = MaterialTheme.typography.headlineSmall)

        Text("رصيد الأعلاف", style = MaterialTheme.typography.titleMedium)

        LazyColumn(
            modifier = Modifier.heightIn(max = 190.dp),
            verticalArrangement = Arrangement.spacedBy(4.dp)
        ) {
            items(feeds) { name ->
                val balance = balances[name] ?: 0.0

                Card(Modifier.fillMaxWidth()) {
                    Row(
                        Modifier.fillMaxWidth().padding(10.dp),
                        horizontalArrangement = Arrangement.SpaceBetween
                    ) {
                        Text(name)
                        Text(
                            "${"%.2f".format(balance)} كغ",
                            color = if (balance <= 0)
                                MaterialTheme.colorScheme.error
                            else MaterialTheme.colorScheme.primary
                        )
                    }
                }
            }
        }

        HorizontalDivider()
        Text("تسجيل استهلاك", style = MaterialTheme.typography.titleMedium)

        DropdownField("نوع العلف", feed, feeds, { feed = it })
        OutlinedTextField(
            quantity, { quantity = it },
            label = { Text("الكمية المستهلكة بالكغ") },
            modifier = Modifier.fillMaxWidth(),
            enabled = !saving
        )

        DropdownField("الفئة", category, categories, { category = it })

        OutlinedTextField(
            count, { count = it },
            label = { Text("عدد الحيوانات") },
            modifier = Modifier.fillMaxWidth(),
            enabled = !saving
        )

        OutlinedTextField(
            notes, { notes = it },
            label = { Text("ملاحظات (اختياري)") },
            modifier = Modifier.fillMaxWidth(),
            enabled = !saving
        )

        Button(
            enabled = !saving,
            onClick = {
                val q = quantity.toDoubleOrNull()
                val c = count.toIntOrNull() ?: 0

                if (q == null || q <= 0 || q.isNaN() || q.isInfinite() || c <= 0) {
                    message = "أدخل كمية وعدداً صحيحين."
                } else {
                    saving = true
                    message = ""

                    onAdd(feed, q, category, c, notes) { success ->
                        saving = false
                        if (success) {
                            quantity = ""
                            notes = ""
                            message = "تم تسجيل الاستهلاك بنجاح."
                        } else {
                            message = "لم يتم التسجيل: الرصيد غير كافٍ أو حدث خطأ."

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun DropdownField(
    label: String,
    value: String,
    options: List<String>,
    onSelect: (String) -> Unit,
    modifier: Modifier = Modifier
) {
    var expanded by remember { mutableStateOf(false) }

    ExposedDropdownMenuBox(
        expanded = expanded,
        onExpandedChange = { expanded = !expanded },
        modifier = modifier
    ) {
        OutlinedTextField(
            value = value,
            onValueChange = {},
            readOnly = true,
            label = { Text(label) },
            trailingIcon = {
                ExposedDropdownMenuDefaults.TrailingIcon(expanded = expanded)
            },
            modifier = Modifier.menuAnchor().fillMaxWidth()
        )

        ExposedDropdownMenu(
            expanded = expanded,
            onDismissRequest = { expanded = false }
        ) {
            options.forEach { option ->
                DropdownMenuItem(
                    text = { Text(option) },
                    onClick = {
                        onSelect(option)
                        expanded = false
                    }
                )
            }
        }
    }
}



@Composable
fun SmartFarmPanel(
    animals: List<Animal>,
    purchases: List<FeedPurchase>,
    consumptions: List<FeedConsumption>,
    tasks: List<FarmTask>,
    events: List<HerdEvent>,
    onAddEvent: (String, String, String, String, (Boolean) -> Unit) -> Unit,
    onAddTask: (String, String, Int, String, (Boolean) -> Unit) -> Unit
) {
    val context = LocalContext.current
    val today = LocalDate.now()
    var section by remember { mutableStateOf("التغذية الذكية") }
    var feedGroup by remember { mutableStateOf("القطيع كله") }
    var dailyPerHead by remember { mutableStateOf("") }
    var targetTag by remember(animals) { mutableStateOf(animals.firstOrNull()?.tag ?: "عام") }
    var eventType by remember { mutableStateOf("تطعيم") }
    var eventDate by remember { mutableStateOf(today.toString()) }
    var eventAmount by remember { mutableStateOf("") }
    var eventNotes by remember { mutableStateOf("") }
    var taskType by remember { mutableStateOf("فحص بيطري") }
    var taskDate by remember { mutableStateOf(today.plusDays(1).toString()) }
    var taskCount by remember(animals.size) { mutableStateOf(animals.size.coerceAtLeast(1).toString()) }
    var taskNotes by remember { mutableStateOf("") }
    var message by remember { mutableStateOf("") }
    var saving by remember { mutableStateOf(false) }

    val selectedAnimals = when (feedGroup) {
        "نعاج" -> animals.filter { it.category.contains("نعجة") || it.category.contains("نعاج") }
        "فحول" -> animals.filter { it.category.contains("فحل") }
        "حملان" -> animals.filter { it.category.contains("حمل") || it.category.contains("خروف") }
        else -> animals
    }
    val daily = (dailyPerHead.replace(',', '.').toDoubleOrNull() ?: 0.0) * selectedAnimals.size
    val totalFeedCost = purchases.sumOf { it.totalPriceDzd ?: 0.0 }
    fun amountFrom(note: String): Double {
        val match = Regex("(?:المبلغ|قيمة البيع|قيمة المصروف|الوزن)[:： ]*([0-9]+(?:[.,][0-9]+)?)")
            .find(note) ?: return 0.0
        return match.groupValues[1].replace(',', '.').toDoubleOrNull() ?: 0.0
    }
    val sales = events.filter { it.eventType == "بيع" }.sumOf { amountFrom(it.notes) }
    val extraExpenses = events.filter { it.eventType == "مصروف" }.sumOf { amountFrom(it.notes) }
    val estimatedBalance = sales - totalFeedCost - extraExpenses
    val upcoming = tasks.mapNotNull { task ->
        val date = runCatching { LocalDate.parse(task.taskDate) }.getOrNull() ?: return@mapNotNull null
        val days = ChronoUnit.DAYS.between(today, date)
        if (!task.completed && days <= 7L) task to days else null
    }.sortedBy { it.second }

    fun csv(value: String) = "\"" + value.replace("\"", "\"\"").replace("\n", " ") + "\""
    val exportText = buildString {
        appendLine("النوع,المعرف,التاريخ,التفاصيل")
        animals.forEach { appendLine("حيوان,${csv(it.tag)},,${csv("${it.sex} - ${it.category} - ${it.weightKg} كغ")}") }
        purchases.forEach { appendLine("شراء علف,${csv(it.feedName)},${csv(it.purchaseDate)},${csv("${it.quantityKg} كغ - ${it.totalPriceDzd ?: 0.0} دج")}") }
        consumptions.forEach { appendLine("استهلاك,${csv(it.feedName)},${csv(it.consumptionDate)},${csv("${it.quantityKg} كغ - ${it.animalCategory}")}") }
        tasks.forEach { appendLine("مهمة,${csv(it.taskType)},${csv(it.taskDate)},${csv(it.notes)}") }
        events.forEach { appendLine("حدث,${csv(it.animalTag)},${csv(it.eventDate)},${csv("${it.eventType} - ${it.notes}")}") }
    }

    Column(verticalArrangement = Arrangement.spacedBy(10.dp)) {
        Text("الإدارة الذكية للمزرعة", style = MaterialTheme.typography.titleLarge,
            color = MaterialTheme.colorScheme.primary)
        Text("أدوات إضافية مدمجة مع بيانات القطيع الحالية.", style = MaterialTheme.typography.bodySmall,
            color = MaterialTheme.colorScheme.onSurfaceVariant)
        DropdownField(
            "القسم الذكي", section,
            listOf("التغذية الذكية", "المرشد العلمي", "الصحة والتناسل", "الوزن والنمو", "المواعيد والتنبيهات", "المالية والتقارير"),
            { section = it }
        )

        when (section) {
            "المرشد العلمي" -> ScientificSheepGuide(animals = animals)
            "التغذية الذكية" -> {
                Text("حاسبة الاحتياج اليومي", style = MaterialTheme.typography.titleMedium)
                DropdownField("الفئة", feedGroup, listOf("القطيع كله", "نعاج", "فحول", "حملان"), { feedGroup = it })
                OutlinedTextField(
                    value = dailyPerHead, onValueChange = { dailyPerHead = it },
                    label = { Text("العلف اليومي للحيوان الواحد بالكغ") },
                    supportingText = { Text("أدخل الحصة التي تعتمدها أنت أو يحددها المختص البيطري.") },
                    modifier = Modifier.fillMaxWidth()
                )
                Card {
                    Column(Modifier.padding(14.dp), verticalArrangement = Arrangement.spacedBy(5.dp)) {
                        Text("عدد الحيوانات: ${selectedAnimals.size}")
                        Text("الاحتياج اليومي: ${"%.2f".format(Locale.US, daily)} كغ")
                        Text("7 أيام: ${"%.1f".format(Locale.US, daily * 7)} كغ")
                        Text("30 يوماً: ${"%.1f".format(Locale.US, daily * 30)} كغ")
                        Text("90 يوماً: ${"%.1f".format(Locale.US, daily * 90)} كغ")
                    }
                }
                Text("مخزون الأعلاف", style = MaterialTheme.typography.titleMedium)
                val feedNames = (purchases.map { it.feedName } + consumptions.map { it.feedName }).distinct().sorted()
                if (feedNames.isEmpty()) Text("لم تسجل مشتريات أو استهلاكاً بعد.")
                feedNames.forEach { name ->
                    val bought = purchases.filter { it.feedName == name }.sumOf { it.quantityKg }
                    val used = consumptions.filter { it.feedName == name }.sumOf { it.quantityKg }
                    ListItem(
                        headlineContent = { Text(name) },
                        supportingContent = { Text("مشتَرى: ${"%.1f".format(Locale.US, bought)} كغ • مستهلك: ${"%.1f".format(Locale.US, used)} كغ") },
                        trailingContent = { Text("${"%.1f".format(Locale.US, bought - used)} كغ",
                            color = if (bought - used <= 0) MaterialTheme.colorScheme.error else MaterialTheme.colorScheme.primary) }
                    )
                    HorizontalDivider()
                }
            }
            "الصحة والتناسل" -> {
                Text("سجل صحي وتناسلي موحد", style = MaterialTheme.typography.titleMedium)
                DropdownField("الحيوان", targetTag, listOf("عام") + animals.map { it.tag }, { targetTag = it })
                DropdownField("نوع الحدث", eventType,
                    listOf("تلقيح", "تطعيم", "علاج", "فحص بيطري", "ولادة", "فطام", "بيع", "مصروف"),
                    { eventType = it })
                OutlinedTextField(eventDate, { eventDate = it }, label = { Text("التاريخ (YYYY-MM-DD)") }, modifier = Modifier.fillMaxWidth())
                if (eventType == "بيع" || eventType == "مصروف") {
                    OutlinedTextField(eventAmount, { eventAmount = it }, label = { Text("المبلغ بالدينار الجزائري") }, modifier = Modifier.fillMaxWidth())
                }
                OutlinedTextField(eventNotes, { eventNotes = it }, label = { Text("التفاصيل والملاحظات") }, modifier = Modifier.fillMaxWidth())
                Button(
                    enabled = !saving,
                    onClick = {
                        val amount = eventAmount.replace(',', '.').toDoubleOrNull()
                        if (runCatching { LocalDate.parse(eventDate) }.isFailure ||
                            ((eventType == "بيع" || eventType == "مصروف") && (amount == null || amount <= 0.0))
                        ) {
                            message = "تحقق من التاريخ والمبلغ."
                        } else {
                            val detail = buildString {
                                if (eventType == "بيع") append("قيمة البيع: $amount دج. ")
                                if (eventType == "مصروف") append("قيمة المصروف: $amount دج. ")
                                append(eventNotes.trim())
                            }
                            saving = true
                            onAddEvent(targetTag, eventType, eventDate, detail) { ok ->
                                saving = false
                                message = if (ok) "تم حفظ السجل." else "تعذر حفظ السجل."
                                if (ok) { eventAmount = ""; eventNotes = "" }
                            }
                        }
                    },
                    modifier = Modifier.fillMaxWidth()
                ) { Text(if (saving) "جارٍ الحفظ..." else "حفظ الحدث") }
                if (message.isNotBlank()) Text(message)
                Text("آخر السجلات", style = MaterialTheme.typography.titleMedium)
                events.take(8).forEach { event ->
                    ListItem(
                        headlineContent = { Text("${event.eventType} • ${event.animalTag}") },
                        supportingContent = { Text("${event.eventDate} — ${event.notes}") }
                    )
                    HorizontalDivider()
                }
            }
            "الوزن والنمو" -> {
                Text("تسجيل متابعة الوزن", style = MaterialTheme.typography.titleMedium)
                DropdownField("الحيوان", targetTag, animals.map { it.tag }, { targetTag = it })
                OutlinedTextField(eventDate, { eventDate = it }, label = { Text("تاريخ القياس") }, modifier = Modifier.fillMaxWidth())
                OutlinedTextField(eventAmount, { eventAmount = it }, label = { Text("الوزن بالكغ") }, modifier = Modifier.fillMaxWidth())
                Button(
                    enabled = !saving && animals.isNotEmpty(),
                    onClick = {
                        val weight = eventAmount.replace(',', '.').toDoubleOrNull()
                        if (weight == null || weight <= 0 || runCatching { LocalDate.parse(eventDate) }.isFailure) {
                            message = "أدخل وزناً وتاريخاً صحيحين."
                        } else {
                            saving = true
                            onAddEvent(targetTag, "وزن", eventDate, "الوزن: $weight كغ") { ok ->
                                saving = false
                                message = if (ok) "تم تسجيل الوزن." else "تعذر تسجيل الوزن."
                                if (ok) eventAmount = ""
                            }
                        }
                    }, modifier = Modifier.fillMaxWidth()
                ) { Text("حفظ قياس الوزن") }
                if (message.isNotBlank()) Text(message)
                Text("سجل الأوزان", style = MaterialTheme.typography.titleMedium)
                events.filter { it.eventType == "وزن" && it.animalTag == targetTag }
                    .sortedByDescending { it.eventDate }.take(12).forEach { item ->
                        ListItem(
                            headlineContent = { Text("${amountFrom(item.notes)} كغ") },
                            supportingContent = { Text(item.eventDate) }
                        )
                    }
            }
            "المواعيد والتنبيهات" -> {
                Text("إضافة موعد أو مهمة", style = MaterialTheme.typography.titleMedium)
                DropdownField("نوع الموعد", taskType,
                    listOf("موعد تلقيح", "ولادة متوقعة", "تطعيم", "علاج", "فحص بيطري", "قص الصوف", "شراء علف", "أخرى"),
                    { taskType = it })
                OutlinedTextField(taskDate, { taskDate = it }, label = { Text("التاريخ (YYYY-MM-DD)") }, modifier = Modifier.fillMaxWidth())
                OutlinedTextField(taskCount, { taskCount = it }, label = { Text("عدد الحيوانات") }, modifier = Modifier.fillMaxWidth())
                OutlinedTextField(taskNotes, { taskNotes = it }, label = { Text("ملاحظات") }, modifier = Modifier.fillMaxWidth())
                Button(
                    enabled = !saving,
                    onClick = {
                        val count = taskCount.toIntOrNull()
                        if (count == null || count <= 0 || runCatching { LocalDate.parse(taskDate) }.isFailure) {
                            message = "تحقق من التاريخ وعدد الحيوانات."
                        } else {
                            saving = true
                            onAddTask(taskType, taskDate, count, taskNotes) { ok ->
                                saving = false
                                message = if (ok) "تمت إضافة الموعد." else "تعذرت إضافة الموعد."
                                if (ok) taskNotes = ""
                            }
                        }
                    }, modifier = Modifier.fillMaxWidth()
                ) { Text("إضافة موعد") }
                if (message.isNotBlank()) Text(message)
                Text("المواعيد المتأخرة والقريبة خلال 7 أيام", style = MaterialTheme.typography.titleMedium)
                if (upcoming.isEmpty()) Text("لا توجد مواعيد متأخرة أو قريبة.")
                upcoming.forEach { (task, days) ->
                    ListItem(
                        headlineContent = { Text(task.taskType) },
                        supportingContent = { Text("${task.taskDate} • ${task.animalCount} حيوان • ${task.notes}") },
                        trailingContent = {
                            Text(if (days < 0) "متأخر ${-days} يوم" else if (days == 0L) "اليوم" else "بعد ${days} يوم",
                                color = if (days < 0) MaterialTheme.colorScheme.error else MaterialTheme.colorScheme.primary)
                        }
                    )
                    HorizontalDivider()
                }
            }
            else -> {
                Text("ملخص مالي تقديري", style = MaterialTheme.typography.titleMedium)
                SummaryCard("تكلفة مشتريات الأعلاف المسجلة", "${"%.0f".format(Locale.US, totalFeedCost)} دج")
                SummaryCard("المصاريف الإضافية المسجلة", "${"%.0f".format(Locale.US, extraExpenses)} دج")
                SummaryCard("إيرادات البيع المسجلة", "${"%.0f".format(Locale.US, sales)} دج")
                SummaryCard("الرصيد التقديري", "${"%.0f".format(Locale.US, estimatedBalance)} دج")
                Text("هذا ملخص للسجلات المدخلة فقط، وليس ربحاً محاسبياً كاملاً.", style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant)
                Button(
                    onClick = {
                        val intent = Intent(Intent.ACTION_SEND).apply {
                            type = "text/csv"
                            putExtra(Intent.EXTRA_SUBJECT, "نسخة بيانات مربي الغنم")
                            putExtra(Intent.EXTRA_TEXT, exportText)
                        }
                        context.startActivity(Intent.createChooser(intent, "تصدير بيانات المزرعة"))
                    }, modifier = Modifier.fillMaxWidth()
                ) { Text("تصدير سجلات المزرعة CSV") }
                Text("يشمل التصدير الحيوانات والمشتريات والاستهلاك والمهام والأحداث المسجلة.", style = MaterialTheme.typography.bodySmall)
            }
        }
    }
}



/**
 * دليل تغذية ورعاية إرشادي للأغنام.
 * القيم تقديرية للمقارنة والتخطيط وليست وصفة علفية مكتملة.
 */
@Composable
fun ScientificSheepGuide(animals: List<Animal>) {
    var selectedTag by remember(animals) { mutableStateOf(animals.firstOrNull()?.tag ?: "") }
    var stage by remember { mutableStateOf("صيانة / نعجة جافة") }
    var lambs by remember { mutableStateOf("واحد") }
    var weightText by remember(animals) {
        mutableStateOf(animals.firstOrNull { it.tag == selectedTag }?.weightKg?.toString() ?: "50")
    }
    var heat by remember { mutableStateOf("معتدل") }
    var daysToLambing by remember { mutableStateOf("30") }

    val selected = animals.firstOrNull { it.tag == selectedTag }
    val weight = weightText.replace(',', '.').toDoubleOrNull()
    val validWeight = weight != null && weight in 3.0..200.0
    val twin = lambs == "توأم"
    val lambCount = if (twin) 2 else 1

    val dmPercent = when (stage) {
        "أواخر الحمل" -> if (twin) 0.026 else 0.025
        "بداية الرضاعة" -> if (twin) 0.030 else 0.028
        "نمو حمل مفطوم" -> 0.035
        "فحل في موسم التلقيح" -> 0.022
        else -> 0.020
    }
    val cpPercent = when (stage) {
        "أواخر الحمل" -> if (twin) 0.13 else 0.11
        "بداية الرضاعة" -> if (twin) 0.16 else 0.14
        "نمو حمل مفطوم" -> 0.16
        "فحل في موسم التلقيح" -> 0.11
        else -> 0.08
    }
    val mePerDay = if (validWeight) {
        val maintenance = 0.394 * Math.pow(weight!!, 0.75)
        when (stage) {
            "أواخر الحمل" -> maintenance + if (twin) 8.0 else 4.0
            "بداية الرضاعة" -> maintenance + if (twin) 19.0 else 12.0
            "نمو حمل مفطوم" -> maintenance * 1.55
            "فحل في موسم التلقيح" -> maintenance + 6.3
            else -> maintenance
        }
    } else 0.0
    val dmKg = if (validWeight) weight!! * dmPercent else 0.0
    val cpGrams = dmKg * cpPercent * 1000.0
    val waterLow = dmKg * 2.1
    val waterHigh = dmKg * 3.1
    val heatFactor = when (heat) {
        "حار" -> 1.25
        "شديد الحرارة" -> 1.50
        else -> 1.0
    }

    Column(verticalArrangement = Arrangement.spacedBy(10.dp)) {
        Text("المرشد العلمي لتغذية ورعاية الأغنام", style = MaterialTheme.typography.titleLarge,
            color = MaterialTheme.colorScheme.primary)
        Text("احسب الاحتياجات حسب الحيوان ووزنه ومرحلة الإنتاج. غيّر الوزن إذا كان القياس أحدث من سجل القطيع.",
            style = MaterialTheme.typography.bodySmall)

        if (animals.isNotEmpty()) {
            DropdownField("الحيوان من سجلات القطيع", selectedTag,
                animals.map { it.tag }, {
                    selectedTag = it
                    weightText = animals.firstOrNull { a -> a.tag == it }?.weightKg?.toString() ?: ""
                })
        } else {
            Text("لا توجد حيوانات مسجلة بعد؛ أدخل الوزن يدوياً لإجراء تقدير عام.")
        }
        OutlinedTextField(
            value = weightText, onValueChange = { weightText = it },
            label = { Text("الوزن الحي بالكيلوغرام") },
            supportingText = { Text("النطاق المقبول للحساب 3 إلى 200 كغ") },
            modifier = Modifier.fillMaxWidth()
        )
        DropdownField("المرحلة الإنتاجية", stage,
            listOf("صيانة / نعجة جافة", "أواخر الحمل", "بداية الرضاعة", "نمو حمل مفطوم", "فحل في موسم التلقيح"),
            { stage = it })
        if (stage == "أواخر الحمل" || stage == "بداية الرضاعة") {
            DropdownField("عدد الحملان", lambs, listOf("واحد", "توأم"), { lambs = it })
        }
        DropdownField("الطقس", heat, listOf("معتدل", "حار", "شديد الحرارة"), { heat = it })
        if (stage == "أواخر الحمل") {
            OutlinedTextField(daysToLambing, { daysToLambing = it.filter(Char::isDigit) },
                label = { Text("الأيام المتبقية للولادة (للتذكير)") },
                modifier = Modifier.fillMaxWidth())
        }

        if (!validWeight) {
            Text("أدخل وزناً صحيحاً قبل عرض الحسابات.", color = MaterialTheme.colorScheme.error)
        } else {
            Card(colors = CardDefaults.cardColors(containerColor = MaterialTheme.colorScheme.surfaceVariant)) {
                Column(Modifier.padding(14.dp), verticalArrangement = Arrangement.spacedBy(7.dp)) {
                    Text("التقدير اليومي للحيوان المحدد", style = MaterialTheme.typography.titleMedium)
                    Text("المادة الجافة: %.2f كغ/يوم".format(Locale.US, dmKg))
                    Text("البروتين الخام المستهدف تقريباً: %.0f غ/يوم".format(Locale.US, cpGrams))
                    Text("الطاقة الأيضية التقديرية: %.1f ميغاجول/يوم".format(Locale.US, mePerDay))
                    Text("ماء الشرب: %.1f – %.1f لتر/يوم".format(Locale.US, waterLow * heatFactor, waterHigh * heatFactor))
                    Text("لـ 7 أيام: %.1f كغ مادة جافة".format(Locale.US, dmKg * 7))
                    Text("لـ 30 يوماً: %.1f كغ مادة جافة".format(Locale.US, dmKg * 30))
                }
            }
            Text("مهم: المادة الجافة ليست وزن العلف كما يقدم. لتحويلها إلى علف فعلي اقسم كمية المادة الجافة على نسبة المادة الجافة المكتوبة أو المحللة للعلف.")
            if (stage == "أواخر الحمل" && daysToLambing.toIntOrNull() in 0..45) {
                Text("تذكير رعاية: خلال آخر 30–45 يوماً قبل الولادة، راجع التحصين المعتمد محلياً وفحص الحالة الجسمية والعليقة مع الطبيب البيطري.",
                    color = MaterialTheme.colorScheme.primary)
            }
        }

        HorizontalDivider()
        Text("المعادن والفيتامينات", style = MaterialTheme.typography.titleMedium)
        Text("• وفر مخلوط أملاح ومعادن مخصصاً للأغنام، مع ماء نظيف باستمرار.")
        Text("• لا تستخدم خلطات الأبقار أو الماعز تلقائياً؛ الأغنام حساسة لتسمم النحاس.")
        Text("• لا تعطِ السيلينيوم أو النحاس أو الفيتامينات بالحقن أو جرعات مركزة دون تشخيص وجرعة يحددها الطبيب/ملصق المنتج.")
        Text("• افحص جودة التبن والعلف والماء؛ تحليل العليقة أفضل من إضافة المكملات عشوائياً.")

        HorizontalDivider()
        Text("خطة وقاية وتلقيح قابلة للتخصيص", style = MaterialTheme.typography.titleMedium)
        listOf(
            "عند إدخال حيوان جديد" to "عزل ومراقبة وفحص بيطري قبل خلطه بالقطيع.",
            "قبل موسم التلقيح" to "فحص الحالة الجسمية والأظلاف والطفيليات والتغذية.",
            "قبل الولادة بـ 30–45 يوماً" to "ناقش جرعة التعزيز ضد المطثيات والكزاز إذا كان ذلك ضمن برنامج المنطقة وسجل التحصين.",
            "الحملان حديثة الولادة" to "تأكد من تناول اللبأ سريعاً، وراقب الحرارة والرضاعة والسرة.",
            "خلال السنة" to "جدول PPR والحمى القلاعية والجدري والطفيليات يحدد وفق تعليمات المصالح البيطرية الجزائرية وخطر المنطقة."
        ).forEach { (heading, detail) ->
            ListItem(headlineContent = { Text(heading) }, supportingContent = { Text(detail) })
            HorizontalDivider()
        }
        Text("لا توجد جرعة لقاح واحدة أو رزنامة ثابتة تصلح لكل المناطق والمنتجات. سجّل اسم اللقاح ورقم التشغيلة وتاريخ الصلاحية، واتبع النشرة والطبيب البيطري.",
            color = MaterialTheme.colorScheme.error, style = MaterialTheme.typography.bodySmall)

        HorizontalDivider()
        Text("متى تطلب الطبيب فوراً؟", style = MaterialTheme.typography.titleMedium)
        Text("توقف الأكل أو الشرب، انتفاخ شديد، صعوبة التنفس، إسهال شديد أو دموي، عرج حاد، حرارة/خمول، أو نعجة حامل لا تستطيع الوقوف.")
        Text("النتائج إرشادية وليست تشخيصاً أو وصفة علفية مكتملة. تتغير الاحتياجات مع السلالة، العمر، الحركة، حالة الجسم، جودة المرعى، المناخ وصحة الحيوان.",
            style = MaterialTheme.typography.bodySmall, color = MaterialTheme.colorScheme.onSurfaceVariant)
    }
}
