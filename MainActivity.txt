class FarmViewModel(private val dao: FarmDao) : ViewModel() {

    val animals = dao.activeAnimals().stateIn(
        viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList()
    )

    val purchases = dao.purchases().stateIn(
        viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList()
    )

    val consumptions = dao.consumptions().stateIn(
        viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList()
    )

    val tasks = dao.tasks().stateIn(
        viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList()
    )

    val events = dao.events().stateIn(
        viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList()
    )

    fun addAnimal(
        tag: String,
        sex: String,
        category: String,
        weight: Double,
        onResult: (String?) -> Unit
    ) {
        viewModelScope.launch {
            try {
                dao.insertAnimal(
                    Animal(
                        tag = tag.trim(),
                        sex = sex,
                        category = category,
                        weightKg = weight
                    )
                )
                onResult(null)
            } catch (_: Exception) {
                onResult("تعذر الحفظ: تأكد أن رمز الحيوان غير مكرر.")
            }
        }
    }

    fun archive(animal: Animal, onResult: (Boolean) -> Unit) {
        viewModelScope.launch {
            try {
                dao.archiveAnimal(animal.tag, "أرشفة يدوية")
                onResult(true)
            } catch (_: Exception) {
                onResult(false)
            }
        }
    }

    fun updateAnimal(
        animal: Animal,
        onResult: (Boolean) -> Unit
    ) {
        viewModelScope.launch {
            try {
                dao.updateAnimal(animal)
                onResult(true)
            } catch (_: Exception) {
                onResult(false)
            }
        }
    }

    fun addAnimalEvent(
        tag: String,
        type: String,
        date: String,
        notes: String,
        onResult: (Boolean) -> Unit
    ) {
        viewModelScope.launch {
            try {
                dao.addEvent(
                    HerdEvent(
                        animalTag = tag,
                        eventType = type,
                        eventDate = date,
                        notes = notes
                    )
                )
                onResult(true)
            } catch (_: Exception) {
                onResult(false)
            }
        }
    }

    fun addPurchase(
        name: String,
        qty: Double,
        price: Double?,
        supplier: String,
        onResult: (Boolean) -> Unit
    ) {
        viewModelScope.launch {
            try {
                dao.addPurchase(
                    FeedPurchase(
                        feedName = name.trim(),
                        quantityKg = qty,
                        totalPriceDzd = price,
                        supplier = supplier,
                        purchaseDate = LocalDate.now().toString()
                    )
                )
                onResult(true)
            } catch (_: Exception) {
                onResult(false)
            }
        }
    }

    fun addConsumption(
        name: String,
        qty: Double,
        category: String,
        count: Int,
        notes: String,
        onResult: (Boolean) -> Unit
    ) {
        viewModelScope.launch {
            try {
                val saved = dao.addConsumptionSafely(
                    FeedConsumption(
                        feedName = name,
                        quantityKg = qty,
                        consumptionDate = LocalDate.now().toString(),
                        animalCategory = category,
                        animalCount = count,
                        notes = notes
                    )
                )
                onResult(saved)
            } catch (_: Exception) {
                onResult(false)
            }
        }
    }

    fun addTask(
        type: String,
        date: String,
        count: Int,
        notes: String,
        onResult: (Boolean) -> Unit
    ) {
        viewModelScope.launch {
            try {
                dao.addTask(
                    FarmTask(
                        taskType = type,
                        taskDate = date,
                        animalCount = count,
                        notes = notes
                    )
                )
                onResult(true)
            } catch (_: Exception) {
                onResult(false)
            }
        }
    }
}

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun FarmScreen(vm: FarmViewModel) {
    val animals by vm.animals.collectAsState()
    val purchases by vm.purchases.collectAsState()
    val consumptions by vm.consumptions.collectAsState()
    val tasks by vm.tasks.collectAsState()
    val events by vm.events.collectAsState()

    var selectedTab by remember { mutableIntStateOf(0) }
    var selectedAnimal by remember { mutableStateOf<Animal?>(null) }
    var screenMessage by remember { mutableStateOf("") }

    val tabs = listOf(
        Triple("الرئيسية", "⌂", 0),
        Triple("القطيع", "🐑", 1),
        Triple("المشتريات", "🛒", 2),
        Triple("التغذية", "🌾", 3),
        Triple("المزيد", "☰", 4)
    )

    Scaffold(
        containerColor = MaterialTheme.colorScheme.background,
        topBar = {
            TopAppBar(
                title = {
                    Column(verticalArrangement = Arrangement.spacedBy(0.dp)) {
                        Text("مربي الغنم", style = MaterialTheme.typography.titleLarge)
                        Text(
                            "إدارة مراحك بسهولة",
                            style = MaterialTheme.typography.labelSmall,
                            color = MaterialTheme.colorScheme.onSurfaceVariant
                        )
                    }
                },
                colors = TopAppBarDefaults.topAppBarColors(
                    containerColor = MaterialTheme.colorScheme.surface
                )
            )
        },
        bottomBar = {
            NavigationBar(
                containerColor = MaterialTheme.colorScheme.surface,
                tonalElevation = 4.dp
            ) {
                tabs.forEachIndexed { index, item ->
                    NavigationBarItem(
                        selected = selectedTab == index && selectedAnimal == null,
                        onClick = {
                            selectedTab = index
                            selectedAnimal = null
                            screenMessage = ""
                        },
                        icon = {
                            Text(
                                item.second,
                                style = MaterialTheme.typography.titleLarge
                            )
                        },
                        label = { Text(item.first, style = MaterialTheme.typography.labelSmall) }
                    )
                }
            }
        }
    ) { padding ->
        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(padding)
                .padding(horizontal = 12.dp, vertical = 8.dp)
        ) {
            if (selectedAnimal != null) {
                val animal = selectedAnimal!!
                AnimalDetailsScreen(
                    animal = animal,
                    events = events.filter { it.animalTag == animal.tag },
                    onBack = { selectedAnimal = null },
                    onUpdate = { updated ->
                        vm.updateAnimal(updated) { success ->
                            screenMessage = if (success) "تم تحديث البيانات بنجاح." else "تعذر تحديث البيانات."
                        }
                    },
                    onAddEvent = { type, date, notes ->
                        vm.addAnimalEvent(animal.tag, type, date, notes) { success ->
                            screenMessage = if (success) "تم تسجيل الحدث." else "تعذر تسجيل الحدث."
                        }
                    },
                    onArchive = {
                        vm.archive(animal) { success ->
                            if (success) selectedAnimal = null
                            else screenMessage = "تعذرت أرشفة الحيوان."
                        }
                    }
                )
                if (screenMessage.isNotBlank()) {
                    Text(
                        screenMessage,
                        modifier = Modifier.padding(8.dp),
                        color = MaterialTheme.colorScheme.primary
                    )
                }
            } else {
                when (selectedTab) {
                    0 -> HomeScreen(
                        animals, purchases, consumptions, tasks, events,
                        onNavigate = { selectedTab = it },
                        onAddEvent = { tag, type, date, notes, result ->
                            vm.addAnimalEvent(tag, type, date, notes, result)
                        },
                        onAddTask = { type, date, count, notes, result ->
                            vm.addTask(type, date, count, notes, result)
                        }
                    )
                    1 -> HerdScreen(
                        animals = animals,
                        onOpen = { selectedAnimal = it },
                        onAdd = { tag, sex, category, weight, result ->
                            vm.addAnimal(tag, sex, category, weight, result)
                        }
                    )
                    2 -> PurchasesScreen(
                        purchases = purchases,
                        onAdd = { name, qty, price, supplier, result ->
                            vm.addPurchase(name, qty, price, supplier, result)
                        }
                    )
                    3 -> FeedingScreen(
                        purchases = purchases,
                        consumptions = consumptions,
                        animals = animals,
                        onAdd = { name, qty, category, count, notes, result ->
                            vm.addConsumption(name, qty, category, count, notes, result)
                        }
                    )
                    else -> SmartFarmPanel(
                        animals = animals,
                        purchases = purchases,
                        consumptions = consumptions,
                        tasks = tasks,
                        events = events,
                        onAddEvent = { tag, type, date, notes, result ->
                            vm.addAnimalEvent(tag, type, date, notes, result)
                        },
                        onAddTask = { type, date, count, notes, result ->
                            vm.addTask(type, date, count, notes, result)
                        }
                    )
                }
            }
        }
    }
}

@Composable
fun HomeScreen(

    tasks: List<FarmTask>,
    events: List<HerdEvent>,
    onNavigate: (Int) -> Unit,
    onAddEvent: (String, String, String, String, (Boolean) -> Unit) -> Unit,
    onAddTask: (String, String, Int, String, (Boolean) -> Unit) -> Unit

    val totalPurchased = purchases.sumOf { it.quantityKg }
    val totalConsumed = consumptions.sumOf { it.quantityKg }

    val upcomingTasks = tasks.filter { task ->
        val date = runCatching { LocalDate.parse(task.taskDate) }.getOrNull()
        date != null && !task.completed && !date.isBefore(today)
    }.sortedBy { it.taskDate }.take(3)

    LazyColumn(
        modifier = Modifier.fillMaxSize(),
        contentPadding = PaddingValues(bottom = 16.dp, top = 4.dp),
        verticalArrangement = Arrangement.spacedBy(14.dp)
    ) {
        item {
            Card(
                modifier = Modifier.fillMaxWidth(),
                shape = MaterialTheme.shapes.extraLarge,
                colors = CardDefaults.cardColors(
                    containerColor = MaterialTheme.colorScheme.primary
                )
            ) {
                Column(
                    Modifier.fillMaxWidth().padding(20.dp),
                    verticalArrangement = Arrangement.spacedBy(7.dp)
                ) {
                    Text(
                        "مرحباً بك في مراحك",
                        color = androidx.compose.ui.graphics.Color.White,
                        style = MaterialTheme.typography.titleMedium
                    )
                    Text(
                        "كل تفاصيل القطيع في مكان واحد",
                        color = androidx.compose.ui.graphics.Color.White,
                        style = MaterialTheme.typography.headlineSmall
                    )

                        "اليوم: ${today}",
                        color = androidx.compose.ui.graphics.Color.White.copy(alpha = 0.82f),
                        style = MaterialTheme.typography.bodySmall
                    )
                }
            }
        }
        item {
            Row(
                Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = androidx.compose.ui.Alignment.CenterVertically
            ) {
                Text("ملخص المراح", style = MaterialTheme.typography.titleLarge,
                    color = MaterialTheme.colorScheme.primary)
                Text("تحديث مباشر", style = MaterialTheme.typography.labelSmall,

        }

            }
        }
        item {
            Text("الوصول السريع", style = MaterialTheme.typography.titleLarge,

        }
        item {

                modifier = Modifier.fillMaxWidth(),
                shape = MaterialTheme.shapes.large,
                colors = CardDefaults.cardColors(containerColor = MaterialTheme.colorScheme.surface)

            }
        }
        item {
            Card(
                modifier = Modifier.fillMaxWidth(),
                colors = CardDefaults.cardColors(containerColor = MaterialTheme.colorScheme.secondaryContainer),
                shape = MaterialTheme.shapes.large
            ) {
                Column(Modifier.padding(16.dp), verticalArrangement = Arrangement.spacedBy(4.dp)) {
                    Text("تذكير للمربي", style = MaterialTheme.typography.titleMedium,
                        color = MaterialTheme.colorScheme.primary)

                }
            }
        }
    }

@Composable
fun HerdScreen(
    animals: List<Animal>,
    onOpen: (Animal) -> Unit,
    onAdd: (String, String, String, Double, (String?) -> Unit) -> Unit
) {
    var tag by remember { mutableStateOf("") }
    var sex by remember { mutableStateOf("أنثى") }
    var category by remember { mutableStateOf("نعجة") }
    var weight by remember { mutableStateOf("") }
    var error by remember { mutableStateOf("") }
    var saving by remember { mutableStateOf(false) }

    Column(modifier = Modifier.fillMaxSize()) {

        Text("القطيع", style = MaterialTheme.typography.headlineSmall)

        OutlinedTextField(
            value = tag,
            onValueChange = { tag = it },
            label = { Text("رقم أو رمز الحيوان") },
            modifier = Modifier.fillMaxWidth(),
            enabled = !saving
        )

        OutlinedTextField(
            value = weight,
            onValueChange = { weight = it },
            label = { Text("الوزن بالكغ") },
            modifier = Modifier.fillMaxWidth(),
            enabled = !saving
        )

        Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
            DropdownField(
                label = "الجنس",
                value = sex,
                options = listOf("أنثى", "ذكر"),
                onSelect = { sex = it },
                modifier = Modifier.weight(1f)
            )
            DropdownField(
                label = "الفئة",
                value = category,
                options = listOf("نعجة", "فحل", "حمل", "خروف"),
                onSelect = { category = it },
                modifier = Modifier.weight(1f)
            )
        }

        Button(
            enabled = !saving,
            onClick = {
                val w = weight.toDoubleOrNull()
                if (tag.isBlank() || w == null || w <= 0 ||
                    w.isNaN() || w.isInfinite()
                ) {
                    error = "أدخل رمز الحيوان ووزناً صحيحاً."
                } else {
                    saving = true
                    error = ""
                    onAdd(tag, sex, category, w) { result ->
                        saving = false
                        if (result == null) {
                            tag = ""
                            weight = ""
                            error = "تمت إضافة الحيوان."
                        } else {
                            error = result
                        }
                    }
                }
            },
            modifier = Modifier.fillMaxWidth()
        ) {
            Text(if (saving) "جارٍ الحفظ..." else "إضافة حيوان")
        }

        if (error.isNotBlank()) {
            Text(error, color = MaterialTheme.colorScheme.error)
        }

        HorizontalDivider()
        Text("الحيوانات النشطة: ${animals.size}")

        LazyColumn(
            modifier = Modifier.weight(1f),
            verticalArrangement = Arrangement.spacedBy(6.dp)
        ) {
            items(animals, key = { it.tag }) { animal ->
                Card(
                    onClick = { onOpen(animal) },
                    modifier = Modifier.fillMaxWidth()
                ) {
                    Column(Modifier.padding(12.dp)) {
                        Text("الرمز: ${animal.tag}")
                        Text("${animal.category} - ${animal.sex}")
                        Text("الوزن: ${animal.weightKg} كغ")
                    }
                }
            }
        }
    }
}

@Composable
fun PurchasesScreen(
    purchases: List<FeedPurchase>,
    onAdd: (String, Double, Double?, String, (Boolean) -> Unit) -> Unit
) {
    val feeds = listOf(
        "الشعير", "التبن", "النخالة",
        "الذرة", "البرسيم الأخضر", "البرسيم المجفف"
    )

    var feed by remember { mutableStateOf(feeds[0]) }
    var quantity by remember { mutableStateOf("") }
    var price by remember { mutableStateOf("") }
    var supplier by remember { mutableStateOf("") }
    var message by remember { mutableStateOf("") }
    var saving by remember { mutableStateOf(false) }

    Column(Modifier.fillMaxSize()) {

        Text("مشتريات الأعلاف", style = MaterialTheme.typography.headlineSmall)

        DropdownField("نوع العلف", feed, feeds, { feed = it })

        OutlinedTextField(
            quantity, { quantity = it },
            label = { Text("الكمية بالكغ") },
            modifier = Modifier.fillMaxWidth(),
            enabled = !saving
        )

        OutlinedTextField(
            price, { price = it },
            label = { Text("السعر الإجمالي بالدينار (اختياري)") },
            modifier = Modifier.fillMaxWidth(),
            enabled = !saving
        )

        OutlinedTextField(
            supplier, { supplier = it },
            label = { Text("المورد (اختياري)") },
            modifier = Modifier.fillMaxWidth(),
            enabled = !saving
        )

        Button(
            enabled = !saving,
            onClick = {
                val q = quantity.toDoubleOrNull()
                val p = if (price.isBlank()) null else price.toDoubleOrNull()

                if (q == null || q <= 0 || q.isNaN() || q.isInfinite() ||
                    (price.isNotBlank() &&
                        (p == null || p < 0 || p.isNaN() || p.isInfinite()))
                ) {
                    message = "تحقق من الكمية والسعر."
                } else {
                    saving = true
                    onAdd(feed, q, p, supplier) { success ->
                        saving = false
                        if (success) {
                            quantity = ""
                            price = ""
                            supplier = ""
                            message = "تم تسجيل الشراء بنجاح."
                        } else {
                            message = "تعذر تسجيل الشراء."
                        }
                    }
                }
            },
            modifier = Modifier.fillMaxWidth()
        ) {
            Text(if (saving) "جارٍ الحفظ..." else "تسجيل الشراء")
        }

        if (message.isNotBlank()) Text(message)

        HorizontalDivider()
        Text("سجل المشتريات", style = MaterialTheme.typography.titleMedium)

        LazyColumn(Modifier.weight(1f)) {
            items(purchases, key = { it.id }) { purchase ->
                ListItem(
                    headlineContent = { Text(purchase.feedName) },
                    supportingContent = {
                        Text("${purchase.quantityKg} كغ - ${purchase.purchaseDate}")
                    },
                    trailingContent = {
                        Text(
                            purchase.totalPriceDzd?.let { "$it دج" }
                                ?: "السعر غير محدد"
                        )
                    }
                )
                HorizontalDivider()
            }
        }
    }
}

@Composable
fun FeedingScreen(

    onAdd: (String, Double, String, Int, String, (Boolean) -> Unit) -> Unit
) {
    val feeds = listOf(
        "الشعير", "التبن", "النخالة",
        "الذرة", "البرسيم الأخضر", "البرسيم المجفف"
    )

    val categories = listOf(
        "القطيع كله", "نعاج", "حوامل",
        "مرضعات", "فحول", "حملان"
    )

    var feed by remember { mutableStateOf(feeds[0]) }
    var category by remember { mutableStateOf(categories[0]) }
    var quantity by remember { mutableStateOf("") }
    var count by remember(animals.size) {
        mutableStateOf(animals.size.toString())
    }
    var notes by remember { mutableStateOf("") }
    var message by remember { mutableStateOf("") }
    var saving by remember { mutableStateOf(false) }

    val balances = feeds.associateWith { name ->
        purchases.filter { it.feedName == name }.sumOf { it.quantityKg } -

        ) {
            items(feeds) { name ->

                Card(Modifier.fillMaxWidth()) {
                    Row(
                        Modifier.fillMaxWidth().padding(10.dp),
                        horizontalArrangement = Arrangement.SpaceBetween
                    ) {
                        Text(name)
                        Text(
                            "${"%.2f".format(balance)} كغ",
                            color = if (balance <= 0)
                                MaterialTheme.colorScheme.error
                            else MaterialTheme.colorScheme.primary

        HorizontalDivider()

            label = { Text("ملاحظات (اختياري)") },
            modifier = Modifier.fillMaxWidth(),
            enabled = !saving
        )

            onClick = {

                            notes = ""

        ) {

        }

                color = if (message.startsWith("تم"))

                else MaterialTheme.colorScheme.error
            )
        }

        Text("سجل الاستهلاك", style = MaterialTheme.typography.titleMedium)

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun DropdownField(
    label: String,
    value: String,
    options: List<String>,
    onSelect: (String) -> Unit,
    modifier: Modifier = Modifier
) {
    var expanded by remember { mutableStateOf(false) }

    ExposedDropdownMenuBox(
        expanded = expanded,
        onExpandedChange = { expanded = !expanded },
        modifier = modifier
    ) {
        OutlinedTextField(
            value = value,
            onValueChange = {},
            readOnly = true,
            label = { Text(label) },
            trailingIcon = {
                ExposedDropdownMenuDefaults.TrailingIcon(expanded = expanded)
            },
            modifier = Modifier.menuAnchor().fillMaxWidth()
        )

        ExposedDropdownMenu(
            expanded = expanded,
            onDismissRequest = { expanded = false }
        ) {
            options.forEach { option ->
                DropdownMenuItem(
                    text = { Text(option) },
                    onClick = {
                        onSelect(option)
                        expanded = false
                    }
                )
            }
        }
    }
}

        tasks.forEach { appendLine("مهمة,${csv(it.taskType)},${csv(it.taskDate)},${csv(it.notes)}") }

    }

    Column(
        Modifier.fillMaxSize().verticalScroll(rememberScrollState()),
        verticalArrangement = Arrangement.spacedBy(12.dp)
    ) {
        if (section == "القائمة") {

/**
 * دليل تغذية ورعاية إرشادي للأغنام.
 * القيم مبنية على جداول إرشادية منشورة، وتظل تقديرية حتى تحليل العلف وتقييم الحيوان.
 */
@Composable
fun ScientificSheepGuide(animals: List<Animal>) {
    var selectedTag by remember(animals) { mutableStateOf(animals.firstOrNull()?.tag ?: "") }
    var stage by remember { mutableStateOf("صيانة / نعجة جافة") }
    var lambs by remember { mutableStateOf("واحد") }
    var weightText by remember(animals) {
        mutableStateOf(animals.firstOrNull { it.tag == selectedTag }?.weightKg?.toString() ?: "50")
    }
    var heat by remember { mutableStateOf("معتدل") }
    var daysToLambing by remember { mutableStateOf("30") }
    var bodyConditionText by remember { mutableStateOf("3") }
    var feedDmText by remember { mutableStateOf("90") }

    val selected = animals.firstOrNull { it.tag == selectedTag }
    val weight = weightText.replace(',', '.').toDoubleOrNull()
    val validWeight = weight != null && weight in 3.0..200.0
    val bcs = bodyConditionText.replace(',', '.').toDoubleOrNull()
    val validBcs = bcs != null && bcs in 1.0..5.0
    val feedDm = feedDmText.replace(',', '.').toDoubleOrNull()
    val validFeedDm = feedDm != null && feedDm in 10.0..100.0
    val twin = lambs == "توأم"

    // Dry-matter intake and crude-protein targets are planning estimates adapted
    // from published ewe tables for a 70 kg reference ewe.
    val dmPercent = when (stage) {
        "أواخر الحمل" -> if (twin) 0.0260 else 0.0257
        "بداية الرضاعة" -> if (twin) 0.0283 else 0.0280
        "نمو حمل مفطوم" -> 0.0300
        "فحل في موسم التلقيح" -> 0.0220
        else -> 0.0169
    }
    val cpPercent = when (stage) {
        "أواخر الحمل" -> if (twin) 0.105 else 0.086
        "بداية الرضاعة" -> if (twin) 0.155 else 0.116
        "نمو حمل مفطوم" -> 0.160
        "فحل في موسم التلقيح" -> 0.095
        else -> 0.075
    }

    val daysLeft = daysToLambing.toIntOrNull()
    val mePerDay = if (validWeight) {
        val maintenance = 0.394 * Math.pow(weight!!, 0.75)
        when (stage) {
            "أواخر الحمل" -> maintenance + if (daysLeft != null && daysLeft <= 14) {
                if (twin) 11.0 else 8.0
            } else {
                if (twin) 5.0 else 4.0
            }
            "بداية الرضاعة" -> maintenance + if (twin) 19.0 else 12.0
            "نمو حمل مفطوم" -> maintenance * 1.55
            "فحل في موسم التلقيح" -> maintenance + 6.3
            else -> maintenance
        }
    } else 0.0

    val dmKg = if (validWeight) weight!! * dmPercent else 0.0
    val cpGrams = dmKg * cpPercent * 1000.0
    val asFedKg = if (validFeedDm && dmKg > 0.0) dmKg / (feedDm!! / 100.0) else 0.0
    val waterLow = dmKg * 2.1
    val waterHigh = dmKg * 3.1
    val heatFactor = when (heat) {
        "حار" -> 1.25
        "شديد الحرارة" -> 1.50
        else -> 1.0
    }

    Column(verticalArrangement = Arrangement.spacedBy(10.dp)) {
        Text("المرشد العلمي لتغذية ورعاية الأغنام", style = MaterialTheme.typography.titleLarge,
            color = MaterialTheme.colorScheme.primary)
        Text(
            "تقدير أولي حسب الوزن والمرحلة الإنتاجية. للحصول على عليقة دقيقة يلزم تحليل الأعلاف وتقييم الحيوان.",
            style = MaterialTheme.typography.bodySmall,
            color = MaterialTheme.colorScheme.onSurfaceVariant
        )

        if (animals.isNotEmpty()) {
            DropdownField("الحيوان من سجلات القطيع", selectedTag,
                animals.map { it.tag }, {
                    selectedTag = it
                    weightText = animals.firstOrNull { a -> a.tag == it }?.weightKg?.toString() ?: ""
                })
            if (selected != null) {
                Text("الفئة المسجلة: \${selected.category} • الجنس: \${selected.sex}",
                    style = MaterialTheme.typography.bodySmall)
            }
        } else {
            Text("لا توجد حيوانات مسجلة؛ أدخل الوزن يدوياً لإجراء تقدير عام.")
        }

        OutlinedTextField(
            value = weightText, onValueChange = { weightText = it },
            label = { Text("الوزن الحي بالكيلوغرام") },
            supportingText = { Text("النطاق المقبول للحساب 3 إلى 200 كغ") },
            modifier = Modifier.fillMaxWidth()
        )
        DropdownField("المرحلة الإنتاجية", stage,
            listOf("صيانة / نعجة جافة", "أواخر الحمل", "بداية الرضاعة", "نمو حمل مفطوم", "فحل في موسم التلقيح"),
            { stage = it })
        if (stage == "أواخر الحمل" || stage == "بداية الرضاعة") {
            DropdownField("عدد الحملان", lambs, listOf("واحد", "توأم"), { lambs = it })
        }
        if (stage == "أواخر الحمل") {
            OutlinedTextField(daysToLambing, { daysToLambing = it.filter(Char::isDigit) },
                label = { Text("الأيام المتبقية للولادة") },
                supportingText = { Text("لتمييز آخر أسبوعين من الحمل") },
                modifier = Modifier.fillMaxWidth())
        }
        OutlinedTextField(
            value = bodyConditionText, onValueChange = { bodyConditionText = it },
            label = { Text("درجة حالة الجسم من 1 إلى 5") },
            supportingText = { Text("1 هزيل جداً • 3 متوسط • 5 سمين جداً") },
            modifier = Modifier.fillMaxWidth()
        )
        DropdownField("الطقس", heat, listOf("معتدل", "حار", "شديد الحرارة"), { heat = it })
        OutlinedTextField(
            value = feedDmText, onValueChange = { feedDmText = it },
            label = { Text("نسبة المادة الجافة في العلف %") },
            supportingText = { Text("قيمة تقريبية من بطاقة العلف أو التحليل؛ التبن والحبوب يختلفان عن السيلاج.") },
            modifier = Modifier.fillMaxWidth()
        )

        } else {

                Column(Modifier.padding(14.dp), verticalArrangement = Arrangement.spacedBy(7.dp)) {

                    Text("المادة الجافة المستهدفة: %.2f كغ/يوم".format(Locale.US, dmKg))

                    Text("ماء الشرب التقريبي: %.1f – %.1f لتر/يوم".format(Locale.US, waterLow * heatFactor, waterHigh * heatFactor))
                    Text("مادة جافة لأسبوع: %.1f كغ".format(Locale.US, dmKg * 7))
                    Text("مادة جافة لشهر: %.1f كغ".format(Locale.US, dmKg * 30))
                }
            }
            if (validFeedDm) {
                Card {
                    Column(Modifier.padding(14.dp), verticalArrangement = Arrangement.spacedBy(6.dp)) {
                        Text("تحويل المادة الجافة إلى وزن العلف المقدم", style = MaterialTheme.typography.titleMedium)
                        Text("إذا كانت العليقة كلها تحتوي على %.0f%% مادة جافة:".format(Locale.US, feedDm!!))
                        Text("%.2f كغ علف كما يقدم يومياً تقريباً".format(Locale.US, asFedKg),
                            style = MaterialTheme.typography.titleLarge, color = MaterialTheme.colorScheme.primary)
                        Text("هذا تحويل حسابي فقط، وليس تحديداً لمكونات العليقة أو نسب الشعير والتبن.")
                    }
                }
            } else {
                Text("أدخل نسبة مادة جافة بين 10 و100% لحساب وزن العلف كما يقدم.",
                    color = MaterialTheme.colorScheme.error)
            }
        }

        if (!validBcs) {
            Text("أدخل درجة حالة جسم بين 1 و5.", color = MaterialTheme.colorScheme.error)
        } else if (bcs!! < 2.0) {
            Text("تنبيه حالة الجسم: الدرجة منخفضة؛ افحص الأسنان والطفيليات والمرض وجودة العلف، وضع خطة تحسين تدريجية مع مختص.",
                color = MaterialTheme.colorScheme.error)
        } else if (bcs!! > 3.5) {
            Text("تنبيه حالة الجسم: الدرجة مرتفعة؛ راجع كثافة الطاقة والحركة لتجنب السمنة ومشكلات الولادة.",
                color = MaterialTheme.colorScheme.primary)
        } else {
            Text("حالة الجسم ضمن النطاق الإرشادي الشائع 2 إلى 3.5؛ واصل المتابعة باللمس وليس بالنظر فقط.",
                color = MaterialTheme.colorScheme.primary)
        }

        HorizontalDivider()
        Text("المعادن والفيتامينات", style = MaterialTheme.typography.titleMedium)
        Text("• وفر ماءً نظيفاً ومخلوط أملاح ومعادن مخصصاً للأغنام باستمرار.")
        Text("• لا تستخدم خلطات الأبقار أو الماعز تلقائياً؛ الأغنام حساسة لتسمم النحاس.")
        Text("• لا تعطِ السيلينيوم أو النحاس أو الفيتامينات بالحقن أو جرعات مركزة دون تشخيص وجرعة يحددها الطبيب البيطري.")
        Text("• افحص جودة التبن والعلف والماء؛ تحليل العليقة أفضل من إضافة المكملات عشوائياً.")

        HorizontalDivider()
        Text("خطة وقاية وتلقيح إرشادية", style = MaterialTheme.typography.titleMedium)
        listOf(
            "عند إدخال حيوان جديد" to "عزل ومراقبة وفحص بيطري قبل خلطه بالقطيع.",
            "قبل موسم التلقيح" to "فحص حالة الجسم والأظلاف والطفيليات والتغذية.",
            "قبل الولادة بـ 30–45 يوماً" to "ناقش جرعة التعزيز ضد المطثيات والكزاز إذا كانت ضمن برنامج الطبيب والمنطقة وسجل التحصين.",
            "الحملان حديثة الولادة" to "تأكد من تناول اللبأ سريعاً، وراقب الحرارة والرضاعة والسرة.",
            "خلال السنة" to "جدول طاعون المجترات الصغيرة والحمى القلاعية والجدري والطفيليات يحدد وفق تعليمات المصالح البيطرية الجزائرية وخطر المنطقة."
        ).forEach { (heading, detail) ->
            ListItem(headlineContent = { Text(heading) }, supportingContent = { Text(detail) })
            HorizontalDivider()
        }
        Text("لا توجد جرعة لقاح أو رزنامة ثابتة تصلح لكل المناطق والمنتجات. سجّل اسم اللقاح ورقم التشغيلة وتاريخ الصلاحية، واتبع النشرة والطبيب البيطري.",
            color = MaterialTheme.colorScheme.error, style = MaterialTheme.typography.bodySmall)

        HorizontalDivider()

        Text("توقف الأكل أو الشرب، انتفاخ شديد، صعوبة التنفس، إسهال شديد أو دموي، عرج حاد، خمول أو حرارة، أو نعجة حامل لا تستطيع الوقوف.")
        Text(
