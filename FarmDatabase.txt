package dz.murabbi.alghanm.data

import android.content.Context
import androidx.room.ColumnInfo
import androidx.room.*
import androidx.room.migration.Migration
import androidx.sqlite.db.SupportSQLiteDatabase
import kotlinx.coroutines.flow.Flow

@Entity(tableName = "animals")
data class Animal(
    @PrimaryKey val tag: String,
    val sex: String,
    val category: String,
    val weightKg: Double,
    val birthDate: String = "",
    val weaningDate: String = "",
    val breed: String = "",
    val photoUri: String = "",
    val motherTag: String = "",
    val fatherTag: String = "",
    val active: Boolean = true,
    val archiveReason: String = "",
    val createdAt: Long = System.currentTimeMillis()
)

@Entity(tableName = "feed_purchases")
data class FeedPurchase(
    @PrimaryKey(autoGenerate = true) val id: Long = 0,
    val feedName: String,
    val quantityKg: Double,
    val totalPriceDzd: Double? = null,
    val supplier: String = "",
    val purchaseDate: String
)

@Entity(tableName = "feed_consumptions")
data class FeedConsumption(
    @PrimaryKey(autoGenerate = true) val id: Long = 0,
    val feedName: String,
    val quantityKg: Double,
    val consumptionDate: String,
    val animalCategory: String = "",
    val animalCount: Int = 0,
    val notes: String = ""
)

@Entity(tableName = "farm_tasks")
data class FarmTask(
    @PrimaryKey(autoGenerate = true) val id: Long = 0,
    val taskType: String,
    val taskDate: String,
    val animalCount: Int,
    val notes: String = "",
    val completed: Boolean = false
)

@Entity(tableName = "herd_events")
data class HerdEvent(
    @PrimaryKey(autoGenerate = true) val id: Long = 0,
    val animalTag: String,
    val eventType: String,
    val eventDate: String,
    val notes: String = ""
)

@Entity(tableName = "vaccine_dose_records")
data class VaccineDoseRecord(
    @PrimaryKey(autoGenerate = true) val id: Long = 0,
    val animalTag: String, val vaccineKey: String, val vaccineName: String,
    val productName: String = "", val administeredDate: String, val doseText: String = "",
    val batchNumber: String = "", val nextDueDate: String = "", val veterinarian: String = "", val notes: String = ""
)

@Entity(tableName = "vaccine_campaigns")
data class VaccineCampaign(
    @PrimaryKey(autoGenerate = true) val id: Long = 0,
    val name: String, val authorityReference: String, val targetGroup: String,
    val startDate: String, val endDate: String = "", val status: String = "معلنة", val notes: String = ""
)

@Entity(tableName = "animal_care_status")
data class AnimalCareStatus(
    @PrimaryKey val animalTag: String,
    val status: String = "طبيعي", val notes: String = "", val updatedAt: Long = System.currentTimeMillis()
)


@Entity(tableName = "treatment_records")
data class TreatmentRecord(
    @PrimaryKey(autoGenerate = true) val id: Long = 0,
    val animalTag: String, val diagnosis: String, val medicineName: String,
    val dose: String = "", val startDate: String, val endDate: String = "",
    val veterinarian: String = "", val withdrawalUntil: String = "", val notes: String = "",
    @ColumnInfo(defaultValue = "0") val quantityUsed: Double = 0.0,
    @ColumnInfo(defaultValue = "0") val medicineBatchId: Long = 0
)

@Entity(tableName = "reproduction_records")
data class ReproductionRecord(
    @PrimaryKey(autoGenerate = true) val id: Long = 0,
    val femaleTag: String, val maleTag: String = "", val eventType: String,
    val eventDate: String, val expectedDate: String = "", val offspringCount: Int = 0,
    val notes: String = "", @ColumnInfo(defaultValue = "''") val offspringTags: String = ""
)

@Entity(tableName = "medicine_stock")
data class MedicineStock(
    @PrimaryKey(autoGenerate = true) val id: Long = 0,
    val medicineName: String, val batchNumber: String = "", val quantity: Double,
    @ColumnInfo(defaultValue = "1.0") val minimumQuantity: Double = 1.0,
    val unit: String = "جرعة", val expiryDate: String, val purchaseDate: String = "",
    val supplier: String = "", val notes: String = ""
)

@Entity(tableName = "audit_entries")
data class AuditEntry(
    @PrimaryKey(autoGenerate = true) val id: Long = 0,
    val action: String, val recordType: String, val recordId: String = "",
    val details: String = "", val createdAt: Long = System.currentTimeMillis()
)

@Entity(tableName = "app_users")
data class AppUser(
    @PrimaryKey val username: String, val role: String,
    val pinHash: String, val active: Boolean = true
)

@Entity(tableName = "finance_entries")
data class FinanceEntry(
    @PrimaryKey(autoGenerate = true) val id: Long = 0,
    val entryType: String,
    val category: String,
    val amountDzd: Double,
    val entryDate: String,
    val description: String = "",
    val animalTag: String = ""
)

@Dao
abstract class FarmDao {

    @Query("SELECT * FROM animals WHERE active = 1 ORDER BY tag")
    abstract fun activeAnimals(): Flow<List<Animal>>

    @Query("SELECT * FROM animals ORDER BY createdAt DESC")
    abstract fun allAnimals(): Flow<List<Animal>>

    @Insert(onConflict = OnConflictStrategy.ABORT)
    abstract suspend fun insertAnimal(animal: Animal)

    @Update
    abstract suspend fun updateAnimal(animal: Animal)

    @Query("UPDATE animals SET active = 0, archiveReason = :reason WHERE tag = :tag")
    abstract suspend fun archiveAnimal(tag: String, reason: String)

    @Insert
    abstract suspend fun addPurchase(purchase: FeedPurchase)

    @Query("SELECT * FROM feed_purchases ORDER BY id DESC")
    abstract fun purchases(): Flow<List<FeedPurchase>>

    @Query("""
        SELECT COALESCE(SUM(quantityKg), 0)
        FROM feed_purchases
        WHERE feedName = :feedName
    """)
    abstract suspend fun purchasedQuantityNow(feedName: String): Double

    @Query("""
        SELECT COALESCE(SUM(quantityKg), 0)
        FROM feed_consumptions
        WHERE feedName = :feedName
    """)
    abstract suspend fun consumedQuantityNow(feedName: String): Double

    @Insert
    protected abstract suspend fun insertConsumption(
        consumption: FeedConsumption
    )

    @Transaction
    open suspend fun addConsumptionSafely(
        consumption: FeedConsumption
    ): Boolean {
        if (consumption.quantityKg <= 0 ||
            consumption.quantityKg.isNaN() ||
            consumption.quantityKg.isInfinite()
        ) return false

        val purchased = purchasedQuantityNow(consumption.feedName)
        val consumed = consumedQuantityNow(consumption.feedName)
        val available = purchased - consumed

        if (consumption.quantityKg > available + 0.000001) {
            return false
        }

        insertConsumption(consumption)
        return true
    }

    @Query("SELECT * FROM feed_consumptions ORDER BY id DESC")
    abstract fun consumptions(): Flow<List<FeedConsumption>>

    @Query("""
        SELECT COALESCE(SUM(quantityKg), 0)
        FROM feed_purchases
        WHERE feedName = :feedName
    """)
    abstract fun purchasedQuantity(feedName: String): Flow<Double>

    @Query("""
        SELECT COALESCE(SUM(quantityKg), 0)
        FROM feed_consumptions
        WHERE feedName = :feedName
    """)
    abstract fun consumedQuantity(feedName: String): Flow<Double>

    @Insert
    abstract suspend fun addTask(task: FarmTask)

    @Query("SELECT * FROM farm_tasks ORDER BY taskDate DESC, id DESC")
    abstract fun tasks(): Flow<List<FarmTask>>

    @Insert
    abstract suspend fun addEvent(event: HerdEvent)

    @Query("SELECT * FROM herd_events ORDER BY id DESC")
    abstract fun events(): Flow<List<HerdEvent>>

    @Query("SELECT * FROM vaccine_dose_records ORDER BY administeredDate DESC, id DESC")
    abstract fun vaccineRecords(): Flow<List<VaccineDoseRecord>>
    @Insert
    abstract suspend fun addVaccineRecord(record: VaccineDoseRecord)
    @Query("SELECT * FROM vaccine_campaigns ORDER BY startDate DESC, id DESC")
    abstract fun vaccineCampaigns(): Flow<List<VaccineCampaign>>
    @Insert
    abstract suspend fun addVaccineCampaign(campaign: VaccineCampaign)
    @Query("SELECT * FROM animal_care_status ORDER BY animalTag")
    abstract fun animalCareStatuses(): Flow<List<AnimalCareStatus>>
    @Insert(onConflict = OnConflictStrategy.REPLACE)
    abstract suspend fun saveAnimalCareStatus(status: AnimalCareStatus)

    @Insert abstract suspend fun addTreatment(record: TreatmentRecord)
    @Update abstract suspend fun updateTreatment(record: TreatmentRecord)
    @Query("DELETE FROM treatment_records WHERE id = :id") abstract suspend fun deleteTreatment(id: Long)
    @Query("SELECT * FROM treatment_records WHERE id = :id LIMIT 1") abstract suspend fun treatmentById(id: Long): TreatmentRecord?
    @Query("UPDATE medicine_stock SET quantity = quantity + :amount WHERE id = :id") abstract suspend fun restoreMedicineStock(id: Long, amount: Double)
    @Query("SELECT * FROM treatment_records ORDER BY startDate DESC, id DESC") abstract fun treatments(): Flow<List<TreatmentRecord>>
    @Insert abstract suspend fun addReproduction(record: ReproductionRecord)
    @Update abstract suspend fun updateReproduction(record: ReproductionRecord)
    @Query("DELETE FROM reproduction_records WHERE id = :id") abstract suspend fun deleteReproduction(id: Long)
    @Query("SELECT * FROM reproduction_records ORDER BY eventDate DESC, id DESC") abstract fun reproductionRecords(): Flow<List<ReproductionRecord>>
    @Query("SELECT * FROM animals WHERE tag = :tag LIMIT 1") abstract suspend fun animalByTag(tag: String): Animal?
    @Query("UPDATE animals SET motherTag = :motherTag, fatherTag = :fatherTag, birthDate = CASE WHEN birthDate = '' THEN :birthDate ELSE birthDate END WHERE tag = :childTag")
    abstract suspend fun linkAnimalParents(childTag: String, motherTag: String, fatherTag: String, birthDate: String): Int
    @Transaction open suspend fun addReproductionWithOffspring(record: ReproductionRecord): Boolean {
        val mother = animalByTag(record.femaleTag) ?: return false
        if (record.eventType == "ولادة") {
            val father = if (record.maleTag.isBlank()) null else animalByTag(record.maleTag) ?: return false
            val tags = record.offspringTags.split(',', '\n', ';').map { it.trim() }.filter { it.isNotBlank() }
            if (record.offspringCount != tags.size || tags.isEmpty() || tags.distinct().size != tags.size) return false
            if (tags.any { it == record.femaleTag || it == record.maleTag || animalByTag(it) == null }) return false
            tags.forEach { child ->
                if (linkAnimalParents(child, mother.tag, father?.tag ?: "", record.eventDate) != 1) return false
            }
        }
        addReproduction(record)
        addAuditEntry(AuditEntry(action = "إضافة", recordType = "تكاثر", details = "${record.femaleTag}: ${record.eventType} • الأبناء: ${record.offspringTags}"))
        return true
    }
    @Insert abstract suspend fun addMedicineStock(item: MedicineStock)
    @Update abstract suspend fun updateMedicineStock(item: MedicineStock)
    @Query("DELETE FROM medicine_stock WHERE id = :id") abstract suspend fun deleteMedicineStock(id: Long)
    @Query("SELECT COUNT(*) FROM treatment_records WHERE medicineBatchId = :id") abstract suspend fun treatmentReferencesBatch(id: Long): Int
    @Query("SELECT COUNT(*) FROM medicine_stock WHERE id = :id") abstract suspend fun medicineBatchExists(id: Long): Int
    @Transaction open suspend fun deleteMedicineStockSafely(id: Long): Boolean {
        if (treatmentReferencesBatch(id) > 0) return false
        deleteMedicineStock(id)
        addAuditEntry(AuditEntry(action = "حذف", recordType = "مخزون دواء", recordId = id.toString()))
        return true
    }
    @Query("SELECT * FROM medicine_stock ORDER BY expiryDate ASC, id DESC") abstract fun medicineStock(): Flow<List<MedicineStock>>
    @Query("SELECT * FROM medicine_stock WHERE medicineName = :name AND quantity >= :amount AND expiryDate >= :today ORDER BY expiryDate ASC, id ASC") abstract suspend fun availableMedicineBatches(name: String, amount: Double, today: String): List<MedicineStock>
    @Query("UPDATE medicine_stock SET quantity = quantity - :amount WHERE id = :id AND quantity >= :amount AND expiryDate >= :today") abstract suspend fun deductMedicineStock(id: Long, amount: Double, today: String): Int
    @Transaction open suspend fun saveTreatmentWithStock(record: TreatmentRecord): Boolean {
        if (record.quantityUsed < 0.0 || record.quantityUsed.isNaN() || record.quantityUsed.isInfinite()) return false
        if (record.quantityUsed > 0.0 && record.medicineName.isBlank()) return false
        val saved = if (record.quantityUsed > 0.0) {
            val today = java.time.LocalDate.now().toString()
            val batch = availableMedicineBatches(record.medicineName, record.quantityUsed, today).firstOrNull() ?: return false
            if (deductMedicineStock(batch.id, record.quantityUsed, today) != 1) return false
            record.copy(medicineBatchId = batch.id)
        } else record
        addTreatment(saved)
        addAuditEntry(AuditEntry(action = "إضافة", recordType = "علاج", details = saved.animalTag + ": " + saved.diagnosis + " • " + saved.medicineName))
        return true
    }

    @Transaction open suspend fun deleteTreatmentAndRestoreStock(id: Long): Boolean {
        val record = treatmentById(id) ?: return false
        if (record.medicineBatchId > 0 && record.quantityUsed > 0.0) {
            if (medicineBatchExists(record.medicineBatchId) != 1) return false
            restoreMedicineStock(record.medicineBatchId, record.quantityUsed)
        }
        deleteTreatment(id)
        addAuditEntry(AuditEntry(action = "حذف", recordType = "علاج", recordId = id.toString(), details = "تمت إعادة كمية الدواء المرتبطة إن وجدت"))
        return true
    }
    @Insert abstract suspend fun addAuditEntry(entry: AuditEntry)
    @Query("SELECT * FROM audit_entries ORDER BY createdAt DESC") abstract fun auditEntries(): Flow<List<AuditEntry>>
    @Insert(onConflict = OnConflictStrategy.REPLACE) abstract suspend fun saveAppUser(user: AppUser)
    @Query("SELECT * FROM app_users WHERE active = 1 ORDER BY username") abstract fun appUsers(): Flow<List<AppUser>>
    @Insert abstract suspend fun addFinanceEntry(entry: FinanceEntry)
    @Update abstract suspend fun updateFinanceEntry(entry: FinanceEntry)
    @Query("DELETE FROM finance_entries WHERE id = :id") abstract suspend fun deleteFinanceEntry(id: Long)
    @Transaction open suspend fun updateFinanceWithAudit(entry: FinanceEntry) {
        updateFinanceEntry(entry)
        addAuditEntry(AuditEntry(action = "تعديل", recordType = "مالية", recordId = entry.id.toString(), details = entry.entryType + " • " + entry.amountDzd + " دج"))
    }
    @Transaction open suspend fun deleteFinanceWithAudit(id: Long) {
        deleteFinanceEntry(id)
        addAuditEntry(AuditEntry(action = "حذف", recordType = "مالية", recordId = id.toString()))
    }
    @Query("SELECT * FROM finance_entries ORDER BY entryDate DESC, id DESC") abstract fun financeEntries(): Flow<List<FinanceEntry>>
    @Query("UPDATE animals SET photoUri = :photoUri WHERE tag = :tag") abstract suspend fun updateAnimalPhoto(tag: String, photoUri: String)

    @Query("SELECT COUNT(*) FROM animals WHERE active = 1")
    abstract fun activeCount(): Flow<Int>
}

@Database(
    entities = [
        Animal::class,
        FeedPurchase::class,
        FeedConsumption::class,
        FarmTask::class,
        HerdEvent::class,
        VaccineDoseRecord::class,
        VaccineCampaign::class,
        AnimalCareStatus::class,
        TreatmentRecord::class, ReproductionRecord::class, MedicineStock::class,
        AuditEntry::class, AppUser::class, FinanceEntry::class
    ],
    version = 8,
    exportSchema = true
)
abstract class FarmDatabase : RoomDatabase() {

    abstract fun farmDao(): FarmDao

    companion object {

        @Volatile
        private var INSTANCE: FarmDatabase? = null

        private val MIGRATION_1_2 = object : Migration(1, 2) {
            override fun migrate(db: SupportSQLiteDatabase) {
                db.execSQL("""
                    CREATE TABLE IF NOT EXISTS feed_consumptions (
                        id INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL,
                        feedName TEXT NOT NULL,
                        quantityKg REAL NOT NULL,
                        consumptionDate TEXT NOT NULL,
                        animalCategory TEXT NOT NULL,
                        animalCount INTEGER NOT NULL,
                        notes TEXT NOT NULL
                    )
                """.trimIndent())
            }
        }

        private val MIGRATION_2_3 = object : Migration(2, 3) {
            override fun migrate(db: SupportSQLiteDatabase) {
                db.execSQL("ALTER TABLE animals ADD COLUMN weaningDate TEXT NOT NULL DEFAULT ''")
                db.execSQL("ALTER TABLE animals ADD COLUMN breed TEXT NOT NULL DEFAULT ''")
                db.execSQL("ALTER TABLE animals ADD COLUMN photoUri TEXT NOT NULL DEFAULT ''")
            }
        }

        private val MIGRATION_3_4 = object : Migration(3, 4) {
            override fun migrate(db: SupportSQLiteDatabase) {
                db.execSQL("CREATE TABLE IF NOT EXISTS vaccine_dose_records (id INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL, animalTag TEXT NOT NULL, vaccineKey TEXT NOT NULL, vaccineName TEXT NOT NULL, productName TEXT NOT NULL, administeredDate TEXT NOT NULL, doseText TEXT NOT NULL, batchNumber TEXT NOT NULL, nextDueDate TEXT NOT NULL, veterinarian TEXT NOT NULL, notes TEXT NOT NULL)")
                db.execSQL("CREATE TABLE IF NOT EXISTS vaccine_campaigns (id INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL, name TEXT NOT NULL, authorityReference TEXT NOT NULL, targetGroup TEXT NOT NULL, startDate TEXT NOT NULL, endDate TEXT NOT NULL, status TEXT NOT NULL, notes TEXT NOT NULL)")
                db.execSQL("CREATE TABLE IF NOT EXISTS animal_care_status (animalTag TEXT NOT NULL PRIMARY KEY, status TEXT NOT NULL, notes TEXT NOT NULL, updatedAt INTEGER NOT NULL)")
            }
        }

        private val MIGRATION_4_5 = object : Migration(4, 5) {
            override fun migrate(db: SupportSQLiteDatabase) {
                db.execSQL("CREATE TABLE IF NOT EXISTS treatment_records (id INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL, animalTag TEXT NOT NULL, diagnosis TEXT NOT NULL, medicineName TEXT NOT NULL, dose TEXT NOT NULL, startDate TEXT NOT NULL, endDate TEXT NOT NULL, veterinarian TEXT NOT NULL, withdrawalUntil TEXT NOT NULL, notes TEXT NOT NULL)")
                db.execSQL("CREATE TABLE IF NOT EXISTS reproduction_records (id INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL, femaleTag TEXT NOT NULL, maleTag TEXT NOT NULL, eventType TEXT NOT NULL, eventDate TEXT NOT NULL, expectedDate TEXT NOT NULL, offspringCount INTEGER NOT NULL, notes TEXT NOT NULL)")
                db.execSQL("CREATE TABLE IF NOT EXISTS medicine_stock (id INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL, medicineName TEXT NOT NULL, batchNumber TEXT NOT NULL, quantity REAL NOT NULL, unit TEXT NOT NULL, expiryDate TEXT NOT NULL, purchaseDate TEXT NOT NULL, supplier TEXT NOT NULL, notes TEXT NOT NULL)")
                db.execSQL("CREATE TABLE IF NOT EXISTS audit_entries (id INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL, action TEXT NOT NULL, recordType TEXT NOT NULL, recordId TEXT NOT NULL, details TEXT NOT NULL, createdAt INTEGER NOT NULL)")
                db.execSQL("CREATE TABLE IF NOT EXISTS app_users (username TEXT NOT NULL PRIMARY KEY, role TEXT NOT NULL, pinHash TEXT NOT NULL, active INTEGER NOT NULL)")
            }
        }

        private val MIGRATION_5_6 = object : Migration(5, 6) {
            override fun migrate(db: SupportSQLiteDatabase) {
                db.execSQL("ALTER TABLE treatment_records ADD COLUMN quantityUsed REAL NOT NULL DEFAULT 0")
                db.execSQL("ALTER TABLE reproduction_records ADD COLUMN offspringTags TEXT NOT NULL DEFAULT ''")
                db.execSQL("CREATE TABLE IF NOT EXISTS finance_entries (id INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL, entryType TEXT NOT NULL, category TEXT NOT NULL, amountDzd REAL NOT NULL, entryDate TEXT NOT NULL, description TEXT NOT NULL, animalTag TEXT NOT NULL)")
            }
        }

        private val MIGRATION_6_7 = object : Migration(6, 7) {
            override fun migrate(db: SupportSQLiteDatabase) {
                db.execSQL("ALTER TABLE treatment_records ADD COLUMN medicineBatchId INTEGER NOT NULL DEFAULT 0")
            }
        }

        private val MIGRATION_7_8 = object : Migration(7, 8) {
            override fun migrate(db: SupportSQLiteDatabase) {
                db.execSQL("ALTER TABLE medicine_stock ADD COLUMN minimumQuantity REAL NOT NULL DEFAULT 1.0")
            }
        }

        fun get(context: Context): FarmDatabase =
            INSTANCE ?: synchronized(this) {
                INSTANCE ?: Room.databaseBuilder(
                    context.applicationContext,
                    FarmDatabase::class.java,
                    "mawashi_pro_v8_standalone20261106.db"
                )
                    .addMigrations(MIGRATION_1_2, MIGRATION_2_3, MIGRATION_3_4, MIGRATION_4_5, MIGRATION_5_6, MIGRATION_6_7, MIGRATION_7_8)
                    .build()
                    .also { INSTANCE = it }
            }
    }
}
